\begin{claim}
Suppose $f$ is a
scalar function
(i.e., ${0\choose 0}$ tensor).
Then

\beq
\call_\vecX f = \vecX(f)
\eeq
\end{claim}
\proof
\qed

\begin{claim}
Suppose $\vecY$ is a
vector field
(i.e., ${1\choose 0}$ tensor).
Then

\beq
\call_\vecX \vecY = 
[\vecX, \vecY]
\eeq
\end{claim}
\proof
\qed


\begin{claim}
Suppose $\pform{\omega}{}$ is a 1-form (i.e., covector, ${0\choose 1}$ tensor).
Then

\beq
\call_\vecX
\ul{\omega}(\vecY)=
\vecX(\ul{\omega}(\vecY))-
\ul{\omega}(
[\vecX, \vecY]
)
\eeq
\end{claim}
\proof
\qed




\begin{claim}
For any smooth tensors
$S$ and $T$,
\beq
\call_\vecX(S\otimes T)=
(\call_\vecX S)\otimes
T
+
S\otimes 
(\call_\vecX T)
\eeq
\end{claim}
\proof
\qed

\begin{claim}
Suppose $T$ is a type
$u\choose d$ tensor. Then

\beq
\ul{\alp}= (\ul{\alp^1},\ul{\alp^2},\cdots, \ul{\alp^u}), \quad
\vecY = (\vecY_1,{\vecY}_2,\cdots,\vecY_d)
\eeq

\beq
(\call_\vecX)
T(\ul{\alp},\vecY)
=
\left\{
\begin{array}{l}
\vecX[
T(\ul{\alp},\vecY)
]
\\
-\sum_{i=1}^u
\AT{T(\ul{\alp},\vecY)}
{\ul{\alp^i}
\rarrow \call_\vecX
\ul{\alp^i}}
\\
-\sum_{i=1}^d
\AT{T(\ul{\alp},\vecY)}
{\vecY_i
\rarrow 
[\vecX, \vecY_i]}
\end{array}
\right.
\eeq
where

\beq
(\call_\vecX\ul{\alp^i})(\vecZ)=
\vecX(\ul{\alp^i}(\vecZ))
-
\ul{\alp^i}
([\vecX, \vecZ])
\eeq


\end{claim}
\proof
\qed

For metric $\metric$
(type ${0\choose 2}$
tensor) 

\beq
\call_\vecX \metric(\vecU, \vecV)
=
\vecX(\metric(\vecU, \vecV))
-
\metric([\vecX, \vecU], \vecV)
-
\metric(\vecU, [\vecX, \vecV])
\eeq
