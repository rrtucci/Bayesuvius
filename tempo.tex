\section{Hodge star operator}

For every $p$-form $\omega(\$^p)$, the the $(n-p)$-form $\star\omega(\$^{n-p})$ is the unique form satisfying

\beq
\alpha\wedge\star\omega
= \av{\alpha,\omega}\text{vol}
\eeq
for every $p$-form $\alpha$.

Here 
$\av{\alpha,\omega}$ is the inner product on $p$-forms induced by the metric.


Suppose
$
\s^1,\ldots,\s^n
$
is an {\bf oriented orthonormal coframe} of 1-forms
Then
\beq
\text{vol}=
\s^1\wedge\cdots\wedge \s^n.
\eeq

The Hodge star is especially simple on basis $p$-forms.
For example,
\beq
\star(\s^{i_1}\wedge\cdots\wedge \s^{i_p})=
\pm \s^{j_1}\wedge\cdots\wedge \s^{j_{n-p}}
\eeq
where the $j$'s are the complementary indices of the $i$'s, and the sign is chosen so that

\beq
(\s^{i_1}\wedge\cdots\wedge \s^{i_p})
\wedge
\star(\s^{i_1}\wedge\cdots\wedge \s^{i_p})=
\text{vol}.
\eeq

Examples
\begin{enumerate}
\item $n=3,\RR^3$

Take the orthonormal basis
$
dx,dy,dz
$
Then

\beq
\star 1
=
dx\wedge dy\wedge dz,
\eeq

\beq
\star dx =
dy\wedge dz,
\eeq
and cyclic permutations of $x,y,z$


\beq
\star(dx\wedge dy)
=
dz
\eeq
and cyclic permutations of $x,y,z$

\item 2-dim polar coordinates.

The coordinate coframe is
$
(dr, d\theta)
$
but these are not orthonormal because
$
g
=
dr^2+r^2d\theta^2$.
Instead define the orthonormal coframe

\beq
\begin{array}{ll}
\s^1=dr,& \check{x}^1 = \hat{r}
\\
\s^2=rd\theta, & \check{x}^2 =\hat{\theta}
\end{array}
\eeq
Now
$g= (1,1)$ so

\beq
\text{vol}=
\s^1\wedge \s^2=
r dr\wedge d\theta
\eeq

\beq
\star \s^1=\s^2, \quad
\star \s^2=-\s^1
\eeq
Hence

\beq
\star dr=rd\theta,\quad
\star d\theta=-\frac{1}{r}dr
\eeq
\end{enumerate}
General coordinate formula

Suppose

\beq
\omega
=
\frac{1}{p!}
\omega_{i_1\cdots i_p}
dx^{i_1}\wedge\cdots\wedge dx^{i_p}
\eeq
Then

\beq
(\star\omega)_{j_1\cdots j_{n-p}}=
\frac{\sqrt{|g|}}{p!}
\omega^{i_1\cdots i_p}
\varepsilon_{i_1\cdots i_pj_1\cdots j_{n-p}}
\eeq
where
$g=\det(g_{ij})$, and $\varepsilon$ is the Levi-Civita symbol.

\section{Expressing grad, div and curl with differential forms}

$\RR^3$, orthonormal coordinates $\hat{x}_i$ ( $=\check{x}_i$ in this case)

\beq
(dx_i(\$))^\sharp = \hatx_i,
\quad
(\hatx_i)^\flat=
dx_i(\$)
\eeq
for $i=1,2,3$

\begin{claim}
\beq
\boxed
{\nabla f =
(\sharp\circ d) [f]}
\eeq
\end{claim}
\proof
\beq
df(\$)=
\sum_{i=1}^3
\partial_i f dx_i(\$)
\eeq
\qed


\begin{claim}
\beq
\boxed
{\nabla \times \vec{V}=
(\sharp\circ \star \circ d \circ \flat)(\vec{V})}
\eeq
\end{claim}
\proof

\beq
\vec{V}= \sum_i
Q_1 \hatx_1
+ Q_2 \hatx_2
+ Q_3 \hatx_3
\eeq

\beq
\omega(\$)=
[\vec{V}]^\flat=
Q_1 dx_1(\$)
+ Q_2 dx_2(\$)
+ Q_3 dx_3(\$)
\eeq

\beq
d\omega(\$^2)=
\left\{
\begin{array}{l}
\left(
\partial_1 Q_2 -\partial_2 Q_1
\right)
dx_1(\$)\wedge dx_2(\$)
\\
\left(
\partial_2 Q_3 -\partial_3 Q_2
\right)
dx_2(\$)\wedge dx_3(\$)
\\
\left(
\partial_3 Q_1 -\partial_1 Q_3
\right)dx_3(\$)\wedge dx_1(\$)
\end{array}
\right.
\eeq


\beq
(\star d\omega)(\$)=
\left\{
\begin{array}{l}
\left(
\partial_1 Q_2 -\partial_2 Q_1
\right)
dx_3(\$)
\\
\left(
\partial_2 Q_3 -\partial_3 Q_2
\right)
dx_1(\$)
\\
\left(
\partial_3 Q_1 -\partial_1 Q_3
\right)dx_2(\$)
\end{array}
\right.
\eeq
\qed


\begin{claim}
\beq
\boxed{
\nabla \cdot \vec{Q}=
(\star\circ
d\circ
\star\circ
\flat)
\vec{Q}
}
\eeq
\end{claim}
\proof
\beq
\omega(\$)=
Q_1 dx_1(\$)
+ Q_2 dx_2(\$)
+ Q_3 dx_3(\$)
\eeq

\beq
(\star\omega)(\$^2)=
Q_1 dx_1(\$)\wedge dx_2(\$)
+ Q_2 dx_2(\$)\wedge dx_3(\$)
+ Q_3 dx_3(\$)\wedge dx_1(\$)
\eeq

\beq
(d\star\omega)(\$^3)=
\underbrace{(\sum_i\partial_i Q_i )}_{\nabla\cdot \vec{Q}}dx_1(\$)\wedge dx_2(\$)\wedge dx_3(\$)
\eeq

\beq
\star(dx_1\wedge  dx_2 \wedge dx_3)= 1
\eeq
\qed

\begin{claim}

\beq
\nabla \times (\nabla f)=0
\eeq

\beq
\nabla \cdot (\nabla \times \vec{V})=0
\eeq
\end{claim}
\proof

\beq
\nabla \times (\nabla f)=
(\sharp\circ \star \circ d \circ \flat)(\sharp\circ d) [f] =0
\eeq
because $d^2=0$.

\beq
\nabla \cdot (\nabla \times \vec{V})=
(\star\circ
d\circ
\star\circ
\flat)(\sharp\circ \star \circ d \circ \flat)\vec{V}=0
\eeq
because $\star\star=\pm 1$ and $d^2=0$

\qed
