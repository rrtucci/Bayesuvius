In the standard Hopf fibration, every pair of distinct fibers is linked exactly once, and no two fibers intersect.
Stereographic projection preserves this property for all rings except those, like the pole ones, which degenerate to straight lines. 


Let

$
S^3=\{(x,y,z,t):\RR^4:
x^2+y^2+z^2 + t^2=1\}=$
surface of a sphere in $\RR^4$, or volume of a sphere in $\RR^3$

$S^3 = \{(\psip, \psim)\in \CC^2 :
|\psip|^2 + |\psim|^2 =1\}$= alternative way of describing $S^3$,
with $\psip=x+iy$, $\psim=z+it$

$
S^2 =\{(x,y,z)\in\RR^3 : x^2 + y^2 + z^2 =1
\}=$ 2-dim surface of a sphere in $\RR^3$

The {\bf Hopf  fibration} $h(\cdot)$ is the map

\beq
{h}: S^3:\rarrow S^2,
\quad
{h}(\psip, \psim)
=\begin{pmatrix}
x\\y\\z
\end{pmatrix}
=
\begin{pmatrix}
2Re(\psip\psim^*)
\\
2Im(\psip \psim^*)
\\
|\psip|^2 -|\psim|^2
\end{pmatrix}
\eeq


\begin{claim}
\beq
(x,y,z)\in S^2
\eeq
\end{claim}
\proof
\beqa
x^2 + y^2 +z^2 
&=&4|\psip\psim^*|^2
+ (|\psip|^2 -|\psim|^2)^2
\\
&=&
(|\psim|^2 + |\psip|^2)^2
\\
&=&
1
\eeqa
\qed

Let

$h^{-1}(x,y,z)=\{(\psipr,\psipi, z')\in S^3 : h(x,y,z)=(\psipr,\psipi,z')\}=$ {\bf fiber}, the inverse image of a point
$(x,y,z)\in S^2$  for a manifold (base space) $S^2$
For some $\alp\in\RR$,

\beq
h(e^{i\alp}\psip,e^{i\alp}\psim)
=h(\psip, \psim)
\eeq
So a curve parametrized by $\alp$
is the inverse image of $(x, y,z)$.

\hrule
Stereographic projection

\beq
\psim =\underbrace{\psimr}_{\lam X} + i \underbrace{\psimi}_{\lam Y}
,\quad
\psip =\underbrace{\psipr}_{\lam Z} + i \underbrace{\psipi}_{\sqrt{1-\lam^2R^2}}
\quad \text{so }
\lam=\frac{1}{R} \sqrt{1-(\psipi)^2}
\eeq

\beq
|\psim|^2+|\psip|^2=1\rarrow
\quad
\lam^2\underbrace{[X^2 +Y^2 + Z^2]}_{R^2}+ 1-\lam^2R^2 =1
\eeq

\beq
\lam=1-\psimi
\eeq
so

\beq
R = \frac{1}{\lam}\sqrt{1-(\psimi)^2}=
\sqrt{\frac{1+\psimi}{1-\psimi}}
\eeq
and, inverting,

\beq
\psimi =\frac{R^2-1}{R^2+1},
\quad \lam=1 -\psimi = \frac{2}{R^2+1}
\eeq
\hrule
Parametrization of fiber


\beq
u^\pm = 
\sqrt{\frac{1\pm z}{2}}
\eeq

\beq
\psim = u^- e^{i\xi}
,\quad
\psip= u^+
e^{i(\xi+\theta)}
\eeq

\beq
x =2 Re(\psip \psim^*)= 2Re(u^+u^-e^{i\theta})=
\underbrace{2u^+u^-}_{\sqrt{1-z^2}}
\cos(\theta)
\eeq

\beq
y = 2Im(\psip \psim^*)= 2Im(u^+u^-e^{i\theta})=
\underbrace{2u^+u^-}_{\sqrt{1-z^2}}\sin(\theta)
\eeq

\beq
z = (u^+)^2 -(u^-)^2= \frac{1+z}{2}-\frac{1-z}{2}=z
\eeq


\beq
\left\{
\begin{array}{l}
\psipr
=u^+\cos(\xi+\theta)
\\
\psipi
=u^+\sin(\xi+\theta)
\\
\psimr
=u^-\cos(\xi)
\\
\psimi
=u^-\sin(\xi)
\end{array}
\right.
\eeq

\beq
\left\{
\begin{array}{l}
X(\xi)=\psipr/(1-\psimi)=
{u^+\cos(\xi+\theta)}/(1-u^-\sin(\xi))
\\
Y(\xi)=\psipi/(1-\psimi)=
{u^+\sin(\xi+\theta)}/(1-u^-\sin(\xi))
\\
Z(\xi)=\psimr/(1-\psimi)=
{u^-\cos(\xi)}/(1-u^-\sin(\xi))
\end{array}
\right.
\eeq

\beq
\lam = 1-u^-\sin \xi
\eeq
\hrule
Phasor picture

\beq
|\psip + \psim|^2 =
\underbrace{|\psip|^2 + |\psim|^2}_1 + \underbrace{\psip\psim^* +\psim\psip^* }_
{=2Re(\psip\psim^*)=x}
\eeq


\beq
|\psip + i\psim|^2 =
\underbrace{|\psip|^2 + |\psim|^2 }_1
+\underbrace{(-i\psip \psim^* +\psim\psip^*)}_{=
  2Im(\psip\psim^*)=y}
\eeq
(sign check: $\psim=1/\sqrt{2}$, $\psip=i/\sqrt{2}$ gives $|\psip+i\psim|^2=2$ and $1+y=2$ )

\begin{claim}
The 
stereographic projection in $\RR^3$ of the fiber
$h^{-1}(x,y,z)$
is a curve $\vgamma(\xi)\in\RR^3$
which we parametrize with $\xi$.

\begin{itemize}
\item
$\vgamma(\xi)=(X(\xi), Y(\xi), Z(\xi))$ lies on the
 surface
of the sphere 

\beq
X^2 + Y^2 + Z^2 = R^2
\eeq
where

\beq R = \sqrt{\frac{1+z}{1-z}}
\eeq

\item
$\vgamma(\xi)$
lies in a plane 
whose normal vector is 
independent of $\xi$
and equals

\beq
\hat{N}=
\frac{1}{
\sqrt{2(1-z)}
}
(x, -y, -(1-z))
\eeq
\item
$\vgamma(\xi)\in\RR^3$
is the circle with center at

\beq C_f=
\frac{1}{1-z}
(y,x, 0)
\eeq
and radius

\beq
r_f = \sqrt{\frac{2}{1-z}}
\eeq

\end{itemize}
 

\end{claim}
\proof

\beq
\psip = \frac{u^+}{u^-}
e^{i\theta}\psim
\eeq

\begin{subequations}
\beq
\psipr = 
\frac{u^+}{u^-}
[\cos\theta\psimr
-\sin\theta\psimi]
\eeq

\beq
\psimi = 
\frac{u^+}{u^-}
[\cos\theta\psimi
+\sin\theta\psimr]
\eeq
\end{subequations}







\begin{subequations}
\beq
Z = 
\frac{u^+}{u^-}
[\cos\theta X
-\sin\theta Y]
\eeq

\beq
\psimi = 
\frac{u^+}{u^-}
[\cos\theta Y
+\sin\theta X](1-\psimi)
\eeq
\end{subequations}



\beq
\frac{u^+}
{u^-}\cos\theta
= \frac{x}{1-z}
,\quad
\frac{u^+}
{u^-}\sin\theta
= \frac{y}{1-z}
\eeq
\hrule

\beq
xX-yY-(1-z)Z=0
\eeq

\beq
\hat{N}\cdot (X,Y,Z)=0
\eeq

\beq
x^2 + y^2 + (1-z)^2
=1 + 1-2z = 2(1-z)
\eeq


\beq
\hat{N}= \frac{1}{\sqrt{2(1-z)}}
(x, -y, -(1-z))
\eeq





\hrule
\beq
\frac{R^2-1}{R^2+1} = 
\left[\frac{x}{1-z} Y
+\frac{y}{1-z} X\right]\frac{2}{R^2+1}
\eeq

\beq
X^2 + Y^2 + Z^2-1-\frac{2x}{1-z}Y -\frac{2y}{1-z}X=0
\eeq

\beq
\left(
X-\frac{y}{(1-z)}
\right)^2
+
\left(
Y-\frac{x}{(1-z)}
\right)
=\underbrace{1 +
\frac{x^2+y^2}{(1-z)^2}}_{r_f^2}
\eeq

\beq
r^2_f=
1 +
\frac{x^2+y^2}{(1-z)^2}=
\frac{(1-z)^2+ 1-z^2}{(1-z)^2}
=\frac{2(1-z)}{(1-z)^2} =\frac{2}
{1-z}
\eeq
\qed
