\begin{claim}
Suppose $x\in\{0,1, 2\} $ and 
$\sum_{x=0}^2P(x)=1$.
Also let $P(x)=p_x$
and

\beq
\metric_{p_{x'} p_{x''}}
=
E
\left[(\partial_{p_{x'}}\ln P(x))
(\partial_{p_{x''}}\ln P(x))\right]
\eeq
Then

\beq
\metric_{p_0 p_0}
=\metric_{p_1 p_1}=
\frac{1}{p_0\para p_2}
\eeq
and

\beq
\metric_{p_1p_0}=
\metric_{p_0p_1} = \frac{1}{p_2}
\eeq
where
\footnote{We are borrowing
notation from Electronics for the resistance $R_1\para R_2$ for 2 resistors $R_1$ and $R_2$ in parallel.}

\beq
p_0\para p_2 =
\left(
\frac{1}{p_0} + \frac{1}{p_2}
\right)^{-1}
\eeq

\end{claim}
\proof


\qed

Note that if $\rvx$ has 2 states instead of 3, we get

\frac{1}{p_0\para p_1}= \frac{1}{p_0p_1}
\eeq
