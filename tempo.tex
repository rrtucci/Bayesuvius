scalar function $f$
( ${0\choose 0}$ tensor)
\beq
\call_\vecX f = \vecX(f)
\eeq

1-form (covector, ${0\choose 1}$ tensor)

\beq
\call_\vecX
\ul{\omega}(\vecY)=
\vecX(\ul{\omega}(\vecY))-
\ul{\omega}(
[\vecX, \vecY]
)
\eeq

Type ${0\choose 2}$ tensor like the metric $\ger{g} $

\beq
\call_\vecX T(\vecU, \vecV)
=
\vecX(T(\vecU, \vecV))
-
T([\vecX, \vecU], \vecV)
-
T(\vecU, [\vecX, \vecV])
\eeq
