Start by parameterizing the surface

\beq
\vecr = \vecr(u_1, u_2)
\eeq
$(u_1, u_2)=(u,v)$


\newcommand{\gauE}{{\av{1|1}}}

\newcommand{\gauF}{{\av{1|2}}}

\newcommand{\gauG}{{\av{1|2}}}

\newcommand{\gaue}{{\av{3|11}}}

\newcommand{\gauf}{{\av{3|12}}}

\newcommand{\gaug}{{\av{3|22}}}


\beq
d\vecr = \partial_{u_1}\vecr\; du_1 +
\partial_{u_2}\vecr\; du_2
\eeq
Latin indices 
will range over
$\{1,2\} $.
$\partial_i =\partial_{u_i}$
Einstein summation convention

\beq
d\vecr = 
\underbrace{\partial_i \vecr}_{\ket{i}}
\;
du_i
\eeq

\beq
\underbrace
{\partial_1 \vecr
\cdot \partial_1\vecr}_
{\gauE}=E,
\quad
\underbrace{\partial_1 \vecr
\cdot \partial_2\vecr}
_{\gauF}=F,
\quad
\underbrace{\partial_2 \vecr
\cdot \partial_2\vecr}_{\gauG}=G
\eeq

First fundamental quadratic form
\beq
Q= d\vecr\cdot d\vecr = du_i du_j \av{i|j}
\eeq

$\hat{n}=\ket{3}$ unit vector
perpendicular to the surface

\beq
\av{3|3}=1,
\quad
\av{3|1}=\av{3|2}=0
\eeq

\beq
\underbrace{\partial_1\partial_1\vecr}_
{\ket{11}}
 = 
\Gamma^1_{11}
\ket{1}
+\Gamma^2_{11}
\ket{2} 
+ e\ket{3}
=
\Gamma^i_{11}\ket{i}+ e \ket{3}
\eeq

\beq
\underbrace{\partial_1\partial_2\vecr}_
{\ket{12}} 
= 
\Gamma^1_{12}
\ket{1}
+\Gamma^2_{12}
\ket{2} 
+ f
\ket{3}
=
\Gamma^i_{12}\ket{i}+ f \ket{3}
\eeq

\beq
\underbrace{\partial_2\partial_2\vecr}
_{\ket{22}}
 = 
\Gamma^1_{22}
\ket{1}
+\Gamma^2_{22}
\ket{2} 
+ g\ket{3}
=
\Gamma^i_{22}\ket{i}+ g \ket{3}
\eeq

\beq
e=\gaue,
\quad
f=\gauf,
\quad
g=\gaug
\eeq

Second Fundamental quadratic form
\beq
Q'= du_i du_j \av{ij|3}
\eeq

Basis $(\ket{1}, \ket{2}, \ket{3})$ is  not 
orthonormal.
Cartan chooses 
orthonormal basis
$(\hate_1, \hate_2, \hate_3)$

\beq
\pform{d\vecr}{} =\pform{ \theta_1}
{}\hate_1
+
\pform{ \theta_2}
{}\hate_2
=
\pform{\theta_i}{}\hate_i
\eeq
