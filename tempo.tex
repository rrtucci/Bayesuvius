\subsection{Torsion}

For a general affine connection (not necessarily
the Levi-Civita one to be discussed later), the {\bf torsion $\TT$}
is defined by

\beq
T(\vecX, \vecY)=
\nnabla_\vecX \vecY  -\nnabla_\vecY \vecX -
[\vecX, \vecY]
\eeq

When $\TT=0$, 

\beq
[\vecX, \vecY] =
\nnabla_\vecX \vecY  -\nnabla_\vecY \vecX
\label{eq-lie-bra-nnablas}
\eeq
If you ever wonder,
can I express a Lie bracket
(which is connection  independent)
in terms of covariant 
derivatives (which embody the connection),
or vice versa, covariant derivatives in terms
of Lie brackets, the answer is yes,
and 
Eq.(\ref{eq-lie-bra-nnablas})
shows how to do it (assuming zero torsion) 



\begin{claim}
In the coordinate basis,

\beq
\TT=0\quad \iff \quad
\Gamma\indices{^\rho_{\mu\nu}}
=
\Gamma\indices{^\rho_{\nu\mu}}
\eeq
\end{claim}
\proof
Set

\beq
\vecX = \vecpder{x^\mu}=\vecpartial_\mu,
\quad
\vecY=
\vecpder{x^\nu}=\vecpartial_\nu
\eeq
Then

\beq
\TT(\vecpartial_\mu,
\vecpartial_\nu)=
\nnabla_{\vecpartial_\mu}
\vec\partial_\nu
-
\nnabla_{\vecpartial_\nu}
\vecpartial_\mu
-
[
\vecpartial_\mu,
\vecpartial_\nu
]
\eeq

By the definition of
the \Chris symbols,

\beq
\nnabla_{\vecpartial_\mu}
\vecpartial_\nu = \Gamma\indices{^\rho_{\mu\nu}}\vecpartial_\rho
\eeq
and the same equation with $\mu\swap \nu$.

For the coordinate basis, 
the basis vectors commute
because partial
derivatives  commute.

\beq
[
\vecpartial_\mu,
\vecpartial_\nu
]=
[\partial_\mu, \partial_\nu]
=0
\eeq

Thus,

\beq
\TT(\vecpartial_\mu,
\vecpartial_\nu)=
\left(
\Gamma\indices{^\rho_{\mu\nu}}-
\Gamma\indices{^\rho_{\nu\mu}}
\right)\vecpartial_\rho
\eeq
\qed

The last 
claim is only valid
in the coordinate basis.
For a general basis (such
as Cartan orthonormal frames) $\vece_\mu$,

\beq
[\vece_\mu, \vece_\nu]\neq0
\eeq
Instead

\beq
[\vece_\mu, \vece_\nu]=
C\indices{^\rho_{\mu\nu}}\vece_\rho
\eeq
Then
\beq
\TT(\vece_\mu, \vece_\nu)=
[\Gamma\indices{^\rho_{\mu\nu}}
-
\Gamma\indices{^\rho_{\nu\mu}}]\vece_\rho
-
[\vece_\mu, \vece_\nu]
\eeq
so


\beq
\TT\indices{^\rho_{\mu\nu}}=
\Gamma\indices{^\rho_{\mu\nu}}
-
\Gamma\indices{^\rho_{\nu\mu}}
-
C\indices{^\rho_{\mu\nu}}
\eeq


\beq
\TT=0\quad \iff
\quad 
\Gamma\indices{^\rho_{\mu\nu}}
-
\Gamma\indices{^\rho_{\nu\mu}}
=
C\indices{^\rho_{\mu\nu}}
\eeq

In Cartan's theory,

\beq
\pform{\TT^a}{2} = \pform{d\theta^a}{2} + \pform{\omega\indices{^a_b}}{}\wedge \pform{\theta^b}{}
\eeq
When $\ul{\theta^a} =\ul{dx^a}$, the 
term $\ul{d\theta^a}$
vanishes.

\subsection{
Levi-Civita connection}


For the {\bf Levi-Civita  connection (LCC) $\nnabla$},
\begin{enumerate}
\item  
The metric is constant
under parallel transport.  This situation
is also referred to as 
{\bf metric compatibility of the connection}
\beq
 \nnabla_\lam\metric_{\mu\nu}=0
\eeq
\item the torsion is zero

\beq 
\TT\indices{^\lam_{\mu\nu}}=0
\eeq
\end{enumerate}

If

\beq
[\nnabla_{\vecpartial_\mu}
\vecpartial_\nu]^\rho=
\Gamma\indices{^\rho_{\mu\nu}}
\eeq
then

\beq
\TT\indices{^\lam_{\mu\nu}}=0
\quad
\text{ iff }\quad
\Gamma\indices{^\rho_{\mu\nu}}
=
\Gamma\indices{^\rho_{\nu\mu}}
\eeq
For LCC,

\beq
[\nnabla_\vecX Y]^\rho=
X^\mu \partial_\mu Y^\rho
+\Gamma\indices{^\rho_{\mu\nu}}
X^\mu
Y^\nu 
\eeq

\begin{claim}
For a { general affine connection $\nnabla$}, in the coordinate basis
\beq
[\call_\vecX \metric]_{\mu\nu}
= 
X^\lam \nnabla_\lam\metric_{\mu\nu}+
\nnabla_\mu X_\nu+
\nnabla_\nu X_\mu
+
X^\lam\left[\TT_{\lam\mu\nu} + (\mu\swap \nu)\right]
\eeq
where $\TT_{\lam \mu\nu}$ represents
the torsion tensor with the $\lam$
index lowered appropriately.
\end{claim}
\proof

The coordinate-free
definition is
\beq
(\call_\vecX \metric)
(\vecU, \vecV)
=
\vecX[\metric(\vecU, \vecV)]
-
\metric([\vecX, \vecU], \vecV)
-
\metric(\vecU,
[\vecX, \vecV])
\eeq

\beq
\vecX[\metric(\vecU, \vecV)]
=
(\nnabla_\vecX\metric)(\vecU, \vecV)
+
\metric)(
\nnabla_\vecX
\vecU, \vecV)
+
\metric)(\vecU, 
\nnabla_\vecX
\vecV)
\eeq

\beq
[\vecX, \vecY]=
\nnabla_\vecX\vecY
-
\nnabla_\vecY \vecX 
-
\TT(\vecX, \vecY)
\eeq
for $\vecY = \vecU, \vecV$.
Substituting, we get

\beq
(\call_\vecX)
(\vecU, \vecV)
=
\left\{
\begin{array}{l}
(\nnabla_\vecX\metric)
(\vecU, \vecV)
\\
+
\metric
(\nnabla_\vecU
\vecX, \vecV)
+
\metric
(\vecU, 
\nnabla_\vecV
\vecX)
\\
+
\metric
(\TT(\vecX,\vecU), \vecV)
+
\metric
(\vecU, 
\TT(\vecX,\vecV))
\end{array}
\right.
\eeq
Now replace

\beq
\vecU=\vecpartial_\mu,
\quad \vecV=\vecpartial_\nu,
\quad
\vecX = X^\lam \vecpartial_\lam
\eeq


\beq
\nnabla_\vecU\vecX=
 \vecpartial_\lam\nnabla_\mu X^\lam
\eeq

\beq
\quad
\metric
(\nnabla_\vecU
\vecX, \vecV)=
\metric_{\lam\nu}
\nnabla_\mu X^\lam=
\nnabla_\mu X_\nu
\eeq

\beq
\TT(\vecX, \vecU)=
X^\lam \TT\indices{^\rho_{\lam\mu}}
\vecpartial_\rho
\eeq

\beq
\metric(\TT(\vecX, \vecU), \vecV)=
X^\lam \TT\indices{^\rho_{\lam\mu}}
 \metric_{\rho\nu}
=X^\lam \TT_{\nu\lam\mu}
\eeq
\qed

For the LLC, the Lie derivative takes this simple form in the coordinate basis,

\beq
[\call_\vecX 
\metric]_{\mu\nu}= \nnabla_\mu X_\nu 
+ (\mu\swap \nu)
\eeq
and the Killing fields satisfy

\beq
0=
[\call_\vecX 
\metric]_{\mu\nu}= 
\nnabla_\mu X_\nu 
+ (\mu\swap \nu)
\eeq

\begin{claim}(Koszul Formula)
\label{cl-Koszul}
If we assume LCC, then
\beq
2\metric(\nnabla_\vecX\vecY, \vecZ)=
\left\{
\begin{array}{l}
\left[\vecX[\metric(\vecY, \vecZ)]
+ (\vecX\swap \vecY)
- (\vecX\swap \vecZ)\right]
\\
-\left[
\metric(\vecX, [\vecY, \vecZ])
+ (\vecX\swap \vecY)
+ (\vecX\swap \vecZ)]
\right]
\end{array}
\right.
\eeq
In the coordinate basis, 

\beq
\Gamma\indices{^\alp_{\mu\nu}}=
\frac{1}{2}
\metric^{\alp\beta}
(
\partial_\mu \metric_{\nu\beta}
+
\partial_\nu \metric_{\mu \beta}
-
\partial_\beta \metric_{\mu\nu}
)
\label{eq-Koszul-index-version}
\eeq

\end{claim}
\proof
$\nnabla \metric=0$ 
for LCC so

\begin{subequations}
\beq
\vecX[\metric(\vecX, \vecZ)]
=
\metric(\nnabla_\vecX \vecY,
\vecZ) 
+
\metric(\vecY, \nnabla_\vecX \vecZ)
\label{eq-kos-sub-formula}
\eeq

\beq
\text{Eq.(\ref{eq-kos-sub-formula}) with }(\vecX\swap\vecY)
\eeq

\beq
\text{Eq.(\ref{eq-kos-sub-formula}) with }
(\vecX\swap\vecZ)
\eeq
\end{subequations}
Adding 
$(a)$ and $(b)$ of Eqs.(\ref{eq-kos-sub-formula})
and subtracting $(c)$,
we get

\beq
\left\{
\begin{array}{l}
\vecX[\metric(\vecY, \vecZ)]
\\
+
(\vecX\swap\vecY)
\\
-
(\vecX\swap\vecZ)
\end{array}
\right\}
=
\left\{
\begin{array}{l}
{\color{red}\metric(\nnabla_\vecX\vecY, \vecZ)}
+
{\color{blue}\metric(\vecY, \nnabla_\vecX\vecZ)}
\\
+{\color{green}\metric(\nnabla_\vecY\vecZ, \vecX)}
+
{\color{red}\metric(\vecZ, \nnabla_\vecY\vecX)}
\\
-
{\color{blue}\metric(\nnabla_\vecZ\vecX, \vecY)}
-
{\color{green}\metric(\vecX, \nnabla_\vecZ\vecY)}
\end{array}
\right\}
\eeq

\beq
\TT(\vecX, \vecY)=
\nnabla_\vecX \vecY
-
\nnabla_\vecY \vecX
-
[\vecX, \vecY]
\eeq
and this equation with swaps
$(\vecX\swap\vecY), (\vecX\swap \vecZ)$
For LCC, $\TT=0$ so

\begin{subequations}
\beq \color{red}
\nnabla_\vecX \vecY
+
\nnabla_\vecY \vecX
=
2\nnabla_\vecY \vecX
+
[\vecX, \vecY]
\eeq

\beq \color{green}
\nnabla_\vecY \vecZ
-
\nnabla_\vecZ \vecY
=
[\vecY, \vecZ]
\eeq

\beq \color{blue}
\nnabla_\vecX \vecZ
-
\nnabla_\vecZ \vecX
=
[\vecX, \vecZ]
\eeq
Finally, assume the coordinate basis.
In that basis, all the Lie brackets
equal zero  because 

\beq
[\vecpartial_\mu, \vecpartial_\nu]=0
\eeq
because partial derivative commute.
Hence, we get 

\beq
2\metric(\nnabla_\vecX\vecY, \vecZ)=
\vecX[\metric(\vecY, \vecZ)]
+ (\vecX\swap \vecY)
- (\vecX\swap \vecZ)
\label{eq-k-pre-coords}
\eeq
Now set

\beq
\vecX =\vecpartial_\mu,
\quad
\vecY =\vecpartial_\nu
,\quad
\vecZ=\vecpartial_\beta
,\quad
[\nnabla_\vecX\vecY]^\alp= \Gamma\indices{^\alp_{\mu\nu}}
\eeq

\beq
2\metric_{\alp\beta}\Gamma\indices{^\alp_{\mu\nu}}
=
\partial_\mu\metric_{\nu\beta}
+(\mu\swap\nu)
-(\mu\swap\beta)
\eeq



\end{subequations}

\qed

The Koszul formula
shows that the LCC 
is unique once
the metric
is specified.
