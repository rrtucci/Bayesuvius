\begin{claim}
\label{cl-fisher-metric-bnet}

Suppose $x\in\{0,1, 2, \ldots, R\} $ and 
$\sum_{x=0}^R P(x)=1$.
Let $P(x)=p_x$
and

\beq
\metric_{p_{x'} p_{x''}}
=
E
\left[(\partial_{p_{x'}}\ln P(x))
(\partial_{p_{x''}}\ln P(x))\right]
\eeq
for $x', x''\in\{ 0, 1, 2\ldots, R\}$
with $p_0, p_1, \ldots, p_{R-1}$ as independent variables and
$p_R= 1-\sum_{i=0}^{R-1}p_i$
as a dependent one.
Then

For $R=1$,

\beq
\metric_{p_0 p_0}=
\frac{1}{p_0\para p_1}= \frac{1}{p_0p_1}
\eeq

For $R=2$,

\beq
\metric_{p_0 p_0}=
\frac{1}{p_0\para p_2}, \quad
\metric_{p_1 p_1}=
\frac{1}{p_1\para p_2}
\eeq
and

\beq
\metric_{p_1p_0}=
\metric_{p_0p_1} = \frac{1}{p_2}
\eeq
where
\footnote{We are borrowing
notation from Electronics for the resistance $R_1\para R_2$ for 2 resistors $R_1$ and $R_2$ in parallel.}

\beq
p_0\para p_2 =
\left(
\frac{1}{p_0} + \frac{1}{p_2}
\right)^{-1}
\eeq

$\metric_{p_{x'} p_{x''}}$ for arbitrary $R$
is evident from the $R=1$ and $R=2$ cases.

\end{claim}
\proof

Consider the $R=2$ case.

\beqa
\partial_{p_0}\ln P(x)
&=&
\partial_{p_0}
\left(
\ln p_0 \delta_x^0 
+ \ln p_1 \delta_x^1
+
\ln(1-p_0-p_1)\delta_x^2
\right)
\\
&=&
\left(
\frac{\delta_x^0}{p_0}
-
\frac{\delta_x^2}{p_2}
\right)
\eeqa

\beq
\metric_{p_0p_0} =
\sum_x p_x\left(
\frac{\delta_x^0}{p_0}
-
\frac{\delta_x^2}{p_2}
\right)^2
=
\frac{p_0}{p_0^2} + \frac{p_2}{p_2^2}
=
\frac{1}{p_0} + \frac{1}{p_2}
\eeq

\beq
\metric_{p_0 p_1}
=
\sum_x p_x 
\left(
\frac{\delta_x^0}{p_0}
-
\frac{\delta_x^2}{p_2}
\right)
\left(
\frac{\delta_x^1}{p_1}
-
\frac{\delta_x^2}{p_2}
\right)
=
\frac{p_2}{p_2^2}
=
\frac{1}{p_2}
\eeq




\qed

{\bf Exponential family
of distributions}

\beq
P(\vecx;\vec{\theta})=
\frac{\phi(\vecx; \vec{\theta})}{Z(\vec{\theta})}
\eeq

probability potential (un-normalized probability)
\beq
\phi(\vec{x};\vec{\theta})=
W(\vec{x})
\exp\left(
\sum_i\theta^i S_i(\vecx)
\right)
\eeq

partition function
\beq
Z(\vec{\theta})=
\sum_{\vecx}\phi(\vecx;\vec{\theta})
\eeq
$S_i(\vecx)$ are called {\bf sufficient statistics}.
