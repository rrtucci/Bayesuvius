\subsection{Four Manifolds}
When expressing Tds in 
Differential Geometry language, we 
consider the following 4 manifolds
\begin{enumerate}
\item $(\calp, \pform{\omega}{2})$= {\bf microscopic symplectic manifold},

coordinates are $\{q^1(i), q^2(i), q^3(i), p_1(i), p_2(i), p_3(i)\}_{i=1}^{N_p}$ so $6N_p$ coordinates. $N_p$ is the number of particles.
If set $q^{a}(i)=q^\mu$ and $p_a(i)= p_\nu$ (so Greek indices label conjugate variable pairs), then

\beq
\ul{\omega} =\sum_\mu \ul{dq^\mu}\wedge \ul{dp_\mu}
\eeq

\item $(\calm_{stat}, \metric_{F})$= {\bf statistical manifold}. This manifold describes a family of
probability distributions. 
If we write the family as $\rho(x; \lam^1, \lam^2, \ldots, \lam^d)$, then its coordinates are the parameters $(\lam^1, \lam^2, \ldots, \lam^d)$ and it has $d$ coordinates.

The {\bf  Fisher Information metric (F metric) }
is 
\beq
\metric^F_{ab}=
\int dx\; \rho(x;\lam)
\pder{\ln\rho}{\lam^a}
\pder{\ln \rho}{\lam^b}
\eeq
\item
$\cale$= {\bf Equilibrium state manifold},

coordinates depend on thermodynamic potential used
(see Table \ref{tab-thermo-pot-ai}). For example, $U=U(S, V, \{N_i\}_{i=1}^{n_c})$ so $2 + 2n_c$ coordinates.
This is called a {\bf state equation}. Regardless of the thermodynamic potential you choose, it's the same manifold,
just expressed in different coordinates.

First Law, at equilibrium:

\beq
dU = TdS -pDV + \sum_i \mu_i dN_i
\eeq

\beq
T = \partial_{S|V, \{N_i\}}U,\quad
p=-\partial_{V|S, \{N_i\}}U,
,\quad
\mu_i =
\partial_{N_i|S,V, \{N_j\}_{j\neq i}}U
\eeq

\item
$(\calt, \pform{\vartheta}{})$= {\bf thermodynamic contact manifold},

coordinates are $(U, S, V, T, P, \{\mu_i\}_{i=1}^{n_c}, \{N_i\}_{i=1}^{n_c})$ so  $5 +2n_c$  coordinates.

The {\bf contact form } $\ul{\vartheta}$ is defined as

\beq
\ul{\vartheta}= \ul{dU}  - T\ul{dS}
+P\ul{dV}
-
\sum_i \mu_i \ul{dN_i}
\eeq
This is before equilibrium. At equilibrium,

\beq
\AT{\ul{\vartheta}}{\cale}=0
\eeq
so the differential
of  the equation of state vanishes.
\end{enumerate}







intensive variables: $p, T, \{\mu_i\}$
$
extensive variables: $U, S, V, \{N_i\}$

$i=1, 2, \cdots, n_c$. $N_c$ is 
the number of components in chemical mixture.

\begin{claim}

\beq
\begin{array}{c}
\xymatrix@C=4pc{
&\ul{dp}\wedge \ul{dS}
\ar[dr]|{-T}
&
\\
\ul{dT}\wedge \ul{dS}
\ar[ur]|{(\partial_T p)_S}
\ar[dr]|{
(\partial_T p)_U(\partial_S U)_T -
(\partial_S p)_T (\partial_T U)_S
}
\ar[rr]|p
&&\Circle{0}
\\
&\ul{dp}\wedge \ul{dU}
\ar[ur]|1
&
}
\\
\\
-T \pderAT{p}{T}{S}
+p
+\left(
\pderAT{p}{T}{S}\pderAT{U}{S}{T}
-\pderAT{p}{S}{T}\pderAT{U}{T}{S}
\right)=0
\end{array}
\eeq

\end{claim}
\proof

\beq
0=
\ul{d^2 V}
=
d\left(
\frac{T}{p}\ul{dS}
-\frac{1}{p}\ul{dU}
\right)=
\frac{1}{p}\ul{dT}\wedge \ul{dS}
-\frac{T}{p^2}\ul{dp}\wedge \ul{dS}
+\frac{1}{p^2}
\ul{dp}\wedge \ul{dU}
\eeq

\beq
0=p \ul{dT}\wedge \ul{dS}
-T \ul{dp}\wedge \ul{dS}
+\ul{dp}\wedge \ul{dU}
\eeq

\beq
\ul{dp}\wedge \ul{dS}=
\pderAT{p}{T}{S}
\ul{dT}\wedge \ul{dS}
\eeq

\beqa
\ul{dp}\wedge \ul{dU}&=&
\left(
\pderAT{p}{T}{S}\ul{dT}
+\pderAT{p}{S}{T}\ul{dS}
\right)
\wedge
\left(
\pderAT{U}{T}{S}\ul{dT}
+\pderAT{U}{S}{T}\ul{dS}
\right)
\\
&=&
\left(
\pderAT{p}{T}{S}\pderAT{U}{S}{T}
-\pderAT{p}{S}{T}\pderAT{U}{T}{S}
\right)
\ul{dT}\wedge \ul{dS}
\eeqa


\qed
