\beq
\dot{\vgamma}(t) = \vecX_{\vgamma(t)}
,\quad
\vgamma(0) =\pi
\eeq

\beq
\vgamma=\invpartial{t}\vecX
\eeq

$PT_{b|a}=$
parallel transport 
of a vector from 
$a\in M$ to $b\in M$



\beq
\boxed
{\AT{\vecX\cdot \nnabla  \vecV}{\pi} =\AT{\nnabla_\vecX\vecV}{\pi}
=
\AT{\frac{d}{dt}}{t=0}
PT(\pi| \vgamma(t)) \vecV_{\vgamma(t)}
\label{eq-gamma-def-nnabla}}
\eeq

Let 

\beq
\vecpartial_i = \vecpder{x_i},
\quad\vecV= V^i
\vecpartial_i
,\quad
\quad\vecX= X^i
\vecpartial_i
\eeq

\begin{claim}

Eq.(\ref{eq-gamma-def-nnabla}) implies

\beq
\nnabla_\vecX \vecV=
X^j
\left(
\partial_j V^i +
\Gamma\indices{^i_{jk}}V^k
\right)
\vecpartial_i
\eeq

\end{claim}
\proof

Suppose vector field
$\vecU$ is tangent to
the curve $\vgamma(t)$:

\beq
\nnabla_{\dot{\vgamma}}\vecU=0
\eeq

\beq
\vecU_{\gamma(t)}=
U^i(\vgamma(t))[\vecpartial_i]_{\vgamma(t)}
\eeq

\beq
\nnabla_{\vecpartial_j}\vecpartial_k=
\nnabla_j\vecpartial_k=
\Gamma\indices{^i_{jk}}\vecpartial_i
\eeq

\beq
\nnabla_{\dot{\vgamma}}\vecpartial_k
=
\dot{\gamma}^j
\nnabla_j\vecpartial_k 
=
\dot{\gamma}^j \Gamma\indices{^i_{jk}}\vecpartial_i
\eeq

\beq
\nnabla_{\dot{\vgamma}}\vecU=
\left(
\frac{dU^i}{dt}
+
\Gamma\indices{^i_{jk}}\dot{\gamma}^j U^k
\right)
\vecpartial_i=0
\eeq
Therefore, parallel transport implies

\beq
\frac{dU^i}{dt}
+
\Gamma\indices{^i_{jk}}\dot{\gamma}^j U^k=0
\eeq
We only need parallel transport
to first order in $t$

\beq
\vecU(0)=\vecV_\pi
\eeq

\beq
\frac{dU(0)}{dt}=-\Gamma\indices{^i_{jk}}X^j_\pi V^k_\pi
\eeq

\beq
PT(\vgamma(t)|\pi)
U^i_\pi
=
U^i(t)\approx V^i_\pi -
t \Gamma\indices{^i_{jk}}X^j_\pi V^k_\pi + \calo(t^2)
\eeq

\beq
PT(\pi|\vgamma(t))
V^i_{\vgamma(t)}
\approx V^i_{\vgamma(t)} +
t \Gamma\indices{^i_{jk}}X^j_{\vgamma(t)} V^k_{\vgamma(t)} + \calo(t^2)
\eeq

\beq
V^i_{\vgamma(t)}
\approx V^i_\pi + t X^i_\pi \partial_j V^i_\pi + \calo(t^2)
 \eeq

\beq
PT(\pi|\vgamma(t))
V^i_{\vgamma(t)}
\approx
V^i_\pi + t X^i_\pi \partial_j V^i_\pi
+
t \Gamma\indices{^i_{jk}}X^j_{\pi} V^k_{\pi} + \calo(t^2)
\eeq

\beqa
\nnabla_\vecX  \vecV
&=&
\lim_{t\rarrow 0}
\frac{PT(\pi|\vgamma(t))\vecV_{\vgamma(t)}- \vecV_\pi}{t}
\\
&=&
\left(X^i \partial_j V^i
+
 \Gamma\indices{^i_{jk}}X^j V^k\right)
\vecpartial_i
\eeqa

\qed
