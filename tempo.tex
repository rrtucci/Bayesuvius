\section{Symplectic Manifold (Hamiltonian Mechanics)}


Symplectic manifolds
find their inspiration 
in Hamiltonian Mechanics (HM),
a sub-field of Classical Mechanics.

A {\bf symplectic manifold} $(M, \ul{\Sigma})$
is defined as a $2\nn$ dimensional manifold $M$ equipped with a 2-form
$\pform{\Sigma}{2}$
which satisfies 2 constraints

\begin{enumerate}
\item closed

\beq
d\ul{\Sigma}=0
\eeq

\item gives a non-degenerate
inner product

\beq
\ul{\Sigma} (\vecX, \vecY)\neq 0
\eeq
for all nonzero $\rvX, \rvY$.
\end{enumerate}

Compare this to a {\bf \Rien manifold} which is equipped with a metric $\metric(\vecX, \vecY)$

\subsection{Darboux Theorem}

\begin{claim}(Darboux Theorem)

Let $(M, \ul{\Sigma}$
be a $2\nn$ dimensional symplectic manifold. 
Then for every point $\pi\in M$,
there exists canonical phase space coordinates 
$(q^i, p_i)_{i=1}^\nn$
(called {\bf Darboux coordinates})
such that
if we define

\beq
\ul{\s}
\eqdef
\sum_i \ul{dq^i}
\wedge \ul{dp_i}
\eeq
then

\beq
\ul{\Sigma}=\ul{\s}
\eeq
\ul{in an open patch
containing $\pi$.}
\end{claim}
\proof
\qed


This means that in the future,
we can assume that
$\ul{\Sigma}=\ul{\s}$.

This also means that 
Symplectic theories are
unique once you fix the manifold dimension $\nn$.
The manifold doesn't have
any distinguishing features
like curvature.

Two simplest examples of symplectic theories are the symplectic plane $(\RR^2, \ul{\s})$ and the 
symplectic sphere $(S^2, \ul{\s})$.
For these 2 cases,
finding a possible
choice of Darboux coordinates is trivial. Darboux Theorem is much more amazing for $\nn>1$ because then
a lot of diagonalization occurs in order to achieve the canonical 2-form  $\ul{\s}$

\begin{enumerate}

\item \begin{claim} A possible choice of Darboux coordinates
for the symplectic plane $(\RR^2, \ul{\s})$
where

\beq
\ul{\s}= f(x, y)\ul{dx}\wedge \ul{dy}
\eeq
where $f(x,y)\neq 0$
is

\beq
q=x,\quad
p(x, y)= \int_{y_0}^y dy'\;
f(x, y')
\eeq
\end{claim}
\proof

\beq
dq = dx
\eeq

\beq
dp= f(x,y) dy
\eeq

\beq
\pder{(q,p)}{(x,y)}=
\left|
\begin{array}{cc}
1&0
\\
\partial_x p & \partial_y p
\end{array}
\right|=\partial_y p=f(x,y)\neq 0
\eeq
\qed


\item 
\begin{claim} A possible choice of Darboux coordinates
for the symplectic sphere $(S^2, \ul{\s})$
where

\beq
\ul{\s}=R^2 \sin\phi\;  \ul{d\phi}\wedge\ul{d\theta}
\eeq
is

\beq
q=\theta,\quad
p=R^2\cos\phi
\eeq
\end{claim}
\proof

\beq
dq = d\theta
\eeq

\beq
dp=-R^2\sin \phi d\phi
\eeq


\qed

Note that these coordinates are not global coordinates for the sphere.
The coordinate $\theta$ fails at the north and south poles, and doesn't return to
its initial value after one turn.

\end{enumerate}

\subsection{Classical Mechanics Review}
This Appendix presents 
some results from
Classical Mechanics 
that are alluded to in this section.
These results were 
discovered without the use of dif-forms
and can be understood 
without them. Dif-forms give a new 
interpretation to them and suggest generalizations of them.

Suppose we have break up the body into $N$ particles with $N$  {\bf positions} $\{\vecr_\alp\}_{\alp=1}^N$ and
$n$  {\bf generalized coordinates }
$\{q^i\}_{i=1}^n$.




\beq
\vecr_\alp=
\vecr_\alp(q^1, q^2, \ldots, q^n)
\eeq
for $\alp=1, 2, \ldots, N$.

$\vec{F}^\infty_\alp=$ {\bf non-contact force}
on particle $\alp$

$\vec{F}^0_\alp=$ {\bf contact force}
on particle $\alp$.
A contact force is a physical force that arises because two bodies are physically touching.
Non-contact forces 
are forces that act at a distance, such as gravity (in the Newtonian description).
Contact forces  are associated with {\bf motion constraints}. For example, if a bead is constrained to a wire, the wire exerts a
contact  force that keeps the bead on the wire. Constraints can be specified thus:


{\bf Newton's Law} for particle $\alp$ says

\beq
m_\alp \ddot{\vecr}_\alp = \vec{F}^\infty_\alp
+ \vec{F}^0_\alp
\eeq


\beq
\vec{F}^0_\alp\cdot
\delta \vecr_\alp=0
\eeq

$\delta \vecr_\alp=$
{\bf virtual displacement} of particle $\alp$.
A virtual displacement
is a man imagined displacement that is not necessarily the displacement that the particle actually makes during the next instant of time.
However, it must be consistent with the constraint.

$\delta W=
\sum_\alp \vec{F}_\alp
\cdot \delta 
\vecr_\alp=$ {\bf virtual work}

{\bf D'Alembert's principle}

\beq
\sum_\alp
(\vec{F}_\alp   -
m_\alp \ddot{r}_\alp)\cdot \delta \vecr_\alp=0
\eeq



$T=$ {\bf kinetic energy}

$V=$ {\bf potential energy}

$L(\vecq, \dot{\vecq}, t)=T-V=$ {\bf Lagrangian}

{\bf Euler-Lagrange's Equations}

\beq
\frac{d}{dt}\pder{L}{\dot{q}}^i -
\pder{L}{q^i}=0
\eeq
for $i=1, 2, \ldots, n$.

$H(\vecq, \vecp, t)=T+V=$ {\bf Hamiltonian}

{\bf Hamilton's equations}

\beq
\dot{q}^i= \pder{H}{p_i},\quad
\dot{p}_i =-\pder{H}{q^i}
\eeq
for $i=1, 2, \ldots, n$.


\begin{claim}

\beq
L_\alp=T_\alp-V_\alp,\quad H_\alp= T_\alp+V_\alp
\eeq
for a point particle $\alp$
for a point particle 
experiencing a conservative force
$-\nabla V_\alp$.
(For multiple particles,  $L=\sum_\alp L$, $H=\sum_\alp H_\alp$)

\end{claim}
\proof 

We'll drop the  particle
subscript $\alp$ in the 
confines of this proof.

The Lagrangian is defined by

\beq
L(\vecr, \vecv)=
T(\vecv)
- V(\vecr)
\eeq
where

\beq
\vecv=\dot{\vecr}
, \quad
T = \frac{1}{2}\vecv^2
\eeq

$L$ is defined this way so that the Euler-Lagrange equations reproduce
Newton's equation.
If $\vecr=(x^1, x^2, x^3)$ and $\vecv = (\dot{x}^1, \dot{x}^2, \dot{x}^3)=(v^1, v^2, v^3)$, then

\beq
\pder{L}{v^i}=
mv_i,
\quad
\pder{L}{x^i}= -\pder{V}{x^i}
\eeq

\beq
0=\frac{d}{dt}
\pder{L}{v^i}-
\pder{L}{x^i}=
m\ddot{x_i}+\pder{V}{x^i}
\eeq 
so

\beq m\ddot{\vecr}= -\nabla V
\eeq
The Hamiltonian
is defined by a Legrendre transform as
\beq
H(\vecr, \vecp)=
\sum _i p_i v^i - L(\vecr, \vecv)
\eeq
where 

\beq
p_i=
\pder{L}{v^i}=
m v^i
\eeq
Therefore

\beqa
H&=& \sum_i m(v^i)^2-L
\\
&=&
\frac{1}{2}m\vecv^2 
+
V 
\\&=&
T+V
\eeqa







\qed


We shall follow the 
historical path:

\beq
\xymatrix@R=1pc{
\text{Newton's Equation}\ar[d]
\\
\text{Virtual work, D'Alembert's principle}\ar[d]
\\\text{Euler-Lagrange Equations}
\ar[d]
\\\text{Hamilton's Equations}\ar[d]
\\
\text{Contact transformations, Poisson brackets}\ar[d]
\\
\text{Symplectic Geometry}
}
\eeq

\begin{claim}
The Euler-Lagrange equations also follow from 
extremizing 
the action.




\end{claim}
\proof

Consider the 
Lagrangian

\beq
L = L(\vecq,
\dot{\vecq}, t)
\eeq
We want to extremize
its action $S(t_2|t_1)$
using calculus of variations].

\beq
S(t_2|t_1)=
\int_{t_1}^{t_2}
dt\;
L(\vecq(t)
, \dot{\vecq}(t), t)
\eeq
We want the physical trajectory $\vecq(t)$ to make $S(t_2|t_1)$ stationary.
Take a one-parameter family of nearby curves

\beq
q^i(t, \eps)=
q^i(t)
+
\eps \eta^i(t)
\eeq
where the endpoints
are fixed:

\beq
\eta^i(t_1)=
\eta^i(t_2)=0
\eeq
Note that

\beq
\pder{{q}^i}{\eps}=
{\eta}^i,\quad
\pder{\dot{q}^i}{\eps}=
\dot{\eta}^i
\eeq
Assume the action is stationary

\beq
0=
\AT{\frac{dS}{d\eps}}{\eps=0}
=
\int_{t_1}^{t_2}dt\;
\AT{\frac{dL}{d\eps}}{\eps=0}=
\int_{t_1}^{t_2}dt\;
\left(
\pder{L}{q^i}
\eta^i
+
\pder{L}{\dot{q}^i}
\dot{\eta}^i
\right)
\eeq

\beq
\pder{L}{\dot{q}^i}
\dot{\eta}^i=
\frac{d}{dt}
\left(
\pder{L}{\dot{q}^i}
{\eta}^i
\right)
-{\eta}^i
\frac{d}{dt}
\pder{L}{\dot{q}^i}
\eeq
Integrate by parts

\beq
0=
\left[
\pder{L}{\dot{q}^i}
{\eta}^i
\right]_{t_1}^{t_2}
+
\int_{t_1}^{t_2}dt\;
\left(
\pder{L}{q^i}
-
\frac{d}{dt}
\pder{L}{\dot{q}^i}
\right)\eta^i
\eeq
Boundary terms vanish because $\eta^i=0$ at endpoints. The variational functions $\eta^i$
are arbitrary so integrand must be zero.


\qed

\subsection{Hamiltonian Theory}

Define {\bf configuration space}
as a manifold $M=\{\{q^i\}_{i=1}^\nn\in \RR^{\nn}\}$ parametrized by $\nn$ position $q^i$.
 
Define {\bf 
phase space}
as a manifold $M=\{(q^i, p_i)\}_{i=1}^\nn\in \RR^{2\nn}\}$ parametrized by $\nn$ position $q^i$, and $\nn$ momenta $p_i$. 

\beq
\pform{\lam}{} =\sum_i q^i \pform{dp_i}{}
\eeq

\beq
\pform{\s}{2}=\pform{d\lam}{2}=
\sum_i \pform{dq^i}{}
\wedge \pform{dp_i}{}
\eeq

\beq
\vecz=
\sum_i
\left[
q^i \vecpder{q^i}+
p_i \vecpder{p_i}
\right]
\eeq

\beq
{\dot{\vecz}}=
\sum_i
\left[
\dot{q}_i \vecpder{q^i} +
\dot{p}_i \vecpder{p_i}
\right]
\label{eq-vecv-def-ham}
\eeq

\begin{claim}
Hamilton's equations given by
\beq
\dot{q}_i=\pder{H}{p_i},
\quad
\dot{p}_i=-\pder{H}{q^i}
\label{eq-ham-eqs}
\eeq
for  $i=1, 2\cdots, \nn$ are satisfied iff

\beq
\begin{array}{cc}
\bcen
\xymatrix{
\pform{dH}{}\ar[d]_1
& \pform{i_{\dot{\vecz}}\s}{}\ar[dl]|1
\\
\Circle{0}&\pform{\s}{2}\ar[u]|{i_{\dot{\vecz}}}
}\ecen
&
\pform{i_{\dot{\vecz}} \s}{}
=
\pform{dH}{}
\end{array}
\eeq

\end{claim}
\proof

\beq
\xymatrix@R=3pc{
\pform{dq}{}
\ar[d]|{\partial_q H}
\ar@/_1pc/[dr]|<<<{\dot{p}}
&\pform{dp}{}
\ar[dl]|{\partial_p H}
\ar[r]|q
\ar[d]|{-\dot{q}}
&\pform{\lam}{}
\ar@/^2pc/[dd]|d
\\
\pform{dH}{}
\ar[d]^1
&\pform{i_{\dot{\vecz}}\s}{}
\ar[dl]|1
\ar[d]_>>>>1
&\pform{i_{\dot{\vecz}}\s}{}
\ar[dl]|1
\\
\Circle{0}
&\Circle{0}
&\pform{\s}{2}
\ar[u]|{i_{\dot{\vecz}}}
}
\eeq


\beq
i_{\dot{\vecz}} \pform{dq^i}{} =\dot{q}_i,
\quad
i_{\dot{\vecz}} \pform{dp_i}{} =\dot{p}_i
\eeq


\beqa
i_{\dot{\vecz}}\pform{\s}{2}
&=&
\sum_i
i_{\dot{\vecz}}[\pform{dq^i}{}\wedge
\pform{dp_i}{}]
\\
&=&
\sum_i
\left[\ul{i_{\dot{\vecz}} dq^i}\wedge \pform{dp_i}{}
-
\pform{dq^i}{}\wedge \ul{i_{\dot{\vecz}} dp_i}
\right]
\\
&=&
\sum_i \left
[\dot{q}_i\pform{dp_i}{}
-
\dot{p}_i \pform{dq^i}{}
\right]
\eeqa

\beq
\pform{dH}{}=\sum_i
\left[
\underbrace{\pder{H}{q^i}}_{-\dot{p}_i}
\pform{dq^i}{}
+
\underbrace{\pder{H}{p_i}}_{\dot{q}^i}
\pform{dp_i}{}
\right]
\eeq

\qed

\subsection{Poisson Brackets}

For a scalar function $f(\;)$,
define the {\bf Hamiltonian vector
field $\hamfield{f}$} by

\beq
i_{\hamfield{f}}\ul{\s} =
\ul{df}
\eeq

Note that 
if ${\dot{\vecz}}$ is vector field defined
in Eq.(\ref{eq-vecv-def-ham}) then Hamilton's equations can be written as

\beq
{\dot{\vecz}}= \hamfield{H}
\eeq

\begin{claim}
In the phase canonical
coordinates,



\beq
\hamfield{f}=
\sum_{i=1}^n
\left(
\pder{f}{p_i}
\vecpder{q^i}
-
\pder{f}{q^i}
\vecpder{p_i
}
\right)
\eeq

\end{claim}
\proof

Let 

\beq
\hamfield{f}=
\sum_i \left(
A^i \vecpder{q^i}
+ B_i \vecpder{p_i}
\right)
\eeq

\beq
\ul{\s}=
\sum_i \ul{dq^i}
\wedge \ul{dp_i}
\eeq
Then

\beqa
i_{\hamfield{f}}
\ul{\s}
&=&
\ul{dq^i}(\hamfield{f})
\ul{dp_i}
-
\ul{dp_i}(\hamfield{f})\ul{dq^i}
\\
&=&
A^i\ul{dp_i}
- B_i \ul{dq^i}
\eeqa
Also

\beq
\ul{df}=
\sum_i\left(
\pder{f}{q^i}
\ul{dq^i}
+
\pder{f}{dp_i}
\ul{p_i}
\right)
\eeq
so

\beq
A^i = \pder{f}{q^i}
,\quad
B_i = -
\pder{f}{p_i}
\eeq

\qed

For two scalar functions $f(\;)$ and $g(\;)$,
define their {\bf Poisson bracket} by

\beq
\{f, g\}=
\sum_{i=1}^\nn
\left(
\pder{f}{q^i}
\pder{g}{p_i}
-
\pder{f}{p_i}
\pder{g}{q^i}
\right)
=
\sum_i
\underbrace{\det \pder{(f, g)}{(q^i, p_i)}}_{\text{det of $2\times 2$ Jacobian Matrix}}
\eeq

Note that $\{f, g\}$ is a sum of determinants of $2\times 2$ matrices, but it's NOT a single determinant. We have two functions, $f$ and $g$,
so to create a square matrix (only square matrices have determinants), we pick a set of 
complementary variables $(q^i,p_i)$ for some $i$. If $\{f, g\}$ were the determinant of a $2\nn\times 2\nn$ matrix,
we could label the \ul{columns} by the $2\nn$ phase space
variables, but then we would only have two labels ($f$ and $g$) for the \ul{rows}.

\begin{claim}

\beq
\{ f, g\}=
\ul{\s}\left(
\hamfield{f}, \hamfield{g}
\right)=

\subsection{Canonical Transformations}

A {\bf canonical transformation} (CT) is
a change of phase space coordinates

\beq
\{(q^i, p_i)\}_{i=1}^\nn
\rarrow \{(Q^i, P_i)\}_{i=1}^\nn
\eeq
that \qt{preserves} the
form of the Hamiltonian
equations.
By this 
we mean that if
the old coordinates
satisfy

\beq
\dot{q}^i = 
\pder{H}{p_i}
,\quad
\dot{p}_i = -\pder{H}{q^i}
\eeq
for $i=1, 2, \ldots,\nn$, 
then the new coordinates 
satisfy

\beq
\dot{Q}^i = 
\pder{K}{P_i}
,\quad
\dot{P}_i = -\pder{K}{Q^i}
\eeq
for $i=1, 2, \ldots,\nn$.
Here $H(\vecq, \vecp)$ is the old Hamiltonian and
$K(\vec{Q}, \vec{P})$ is the new one.



\hrule
Notation so far: $\vec{A} = A^i \vecpartial_{x^i}=A^i\partial_i$

 

New notation:

 $\colvec{A}=\begin{bmatrix}A^1\\A^2\\\vdots\\ A^n\end{bmatrix}$: {\bf column vector} in  matrix equations

 $\colvec{A}^T:$ {\bf row vector} in matrix equations.

$\partial_\colvec{x}=\begin{bmatrix}\partial_1\\\partial_2\\\vdots\\ \partial_n\end{bmatrix}$,

$\partial_\colvec{A} = \colvec{A}^T \partial_{\colvec{x}}$




\beq
\colvec{z} = \begin{pmatrix}
\colvec{p}
\\
\colvec{q}
\end{pmatrix}
,\quad
\colvec{Z} = \begin{pmatrix}
\colvec{Q}
\\
\colvec{P}
\end{pmatrix}
\eeq



\beq
\text{Jacobian matrix:}\quad
J_{Z/z}=\pder{\colvec{Z}}{\colvec{z}}
=
\begin{pmatrix}
\pder{Z^1}{z^1}&\cdots&\pder{Z^1}{z^\nn}
\\
\vdots&\vdots&\vdots
\\
\pder{Z^\nn}{z^1}&\cdots&\pder{Z^\nn}{z^\nn}
\end{pmatrix}
\eeq
Note our convention
for translating a Jacobian matrix from index to matrix notation: In your head, rewrite
$\pder{x}{y}$ as $\partial x/\partial y$. This suggests that $x_i$ label the rows and $y_j$ label the columns.

Note that

\beq
J_{Z/z}J_{z/Z}=J_{z/Z}J_{Z/z}=1
\eeq
so

\beq
J^{-1}_{z/Z}= J_{Z/z}, \quad
J^{-1}_{Z/z} = J_{z/Z}
\eeq

For an orthogonal transformation (preserves angles and lengths)
\beq
J_{Z/z}J^T_{Z/z}=J_{z/Z}J^T_{z/Z}=1
\eeq

\beq
(\det J_{Z/z})^2 = (\det J_{z/Z})^2 =1
\quad \implies 
\det(J_{Z/z})=\pm 1
\eeq
Actually, we refer to a transformation with $\det J = 1$ as orthogonal and 
one with $\det J=-1$  as a reflection.

For a symplectic transformation (preserves certain areas):

\beq
J_{Z/z} A J_{Z/z} = J_{z/Z}A J_{z/Z}=A
\eeq
Just as in the orthogonal case,

\beq
(\det J_{Z/z})^2 = (\det J_{z/Z})^2 =1
\quad \implies 
\det(J_{Z/z})=\pm 1
\eeq
Actually, we refer to a transformation with $\det J = 1$ as symplectic and 
one with $\det J=-1$  as anti-symplectic.

A general coordinate transformation need 
not be either orthogonal nor symplectic. In general, $J_{Z/z} J_{z/Z}$ is not necessarily $1$. However,
canonical transformations are always required to be 
symplectic transformations.

An example of a transformation that is symplectic but not orthogonal is

\beq
J_{Z/z}=
\begin{pmatrix}
a&0
\\
0& \frac{1}{a}
\end{pmatrix}
\eeq

\colvec{Z}= J_{Z/z}\colvec{z},
\quad
\colvec{z}= J_{z/Z}\colvec{Z}
\eeq

\quad

 \colvec{\partial}_Z
\eeq

transformations of vectors and covectors. See Section \ref{sec-coord-trans})

\begin{claim}
The  following statements are equivalent
\begin{enumerate}
\item \label{enum-symp-can}
Transformation $\colvec{z}\rarrow \colvec{Z}$ is canonical
(i.e., preserves the form of Hamilton's equations)
\item \label{enum-symp-jaj}
\beq
J_{Z/z} A J^T_{Z/z}=A,
\quad \det J=+1
\eeq
This means $J\in Sp(2\nn, \RR)$ 


\item \label{enum-symp-2form}
\beq
\sum_i \ul{dq^i}\wedge\ul{dp_i}=  \sum_\alp \ul{dQ^\alp}\wedge \ul{dP_\alp}
\eeq
\end{enumerate}
\end{claim}
\proof

\ref{enum-symp-can} $\iff$ \ref{enum-symp-jaj}:

\beqa
\colvec{\dot{Z}}&=&
J_{Z/z}\colvec{\dot{z}}
\\&=&
J_{Z/z} [A\colvec{\partial}_zH]
\\&=&
J_{Z/z}A [J^T_{Z/z}\colvec{\partial}_Z]H
\\
&=&
J_{Z/z}A J^T_{Z/z}\colvec{\partial}_ZK
\eeqa


\beq
\colvec{\dot{Z}}=
A\colvec{\partial}_ZK
\eeq


\beq
J_{Z/z}A J^T_{Z/z}=A
\eeq

\ref{enum-symp-jaj} $\iff$
\ref{enum-symp-2form} :

\beqa
\ul{\omega}_{q,p}
&=&
\sum_i \ul{dq^i}
\wedge
\ul{dp_i}
\\
&=&
\sum_i 
\left(
\ul{d\colvec{q}}^T\ul{d\colvec{p}}-
\ul{d\colvec{p}}^T\ul{d\colvec{q}}
\right)
\\
&=&
(\ul{\colvec{dz}})^T
A\ul{\colvec{dz}}
\eeqa

\beqa
\ul{\omega}_{Q,P} &=&
(\ul{\colvec{dZ}})^T
A\ul{\colvec{dZ}}
\\
&=&
(J_{Z/z}\ul{\colvec{dz}})^T
AJ_{Z/z}\ul{\colvec{dz}}
\\
&=&
(\ul{\colvec{dZ}})^T
J^T_{Z/z}A J_{Z/z}
A\ul{\colvec{dZ}}
\eeqa

\qed

preserves the Poisson brackets.

\end{claim}

\sum_i
\det\left(
\pder{(f, g)}{(q^i, p_i)}J_{z^i/Z^i}
\right)
\\
&=&
\underbrace{\det \left(\pder{(f, g)}{(q^i, p_i)}\right)}_{\{f,g\}_{p, q}}
\underbrace{\det J_{z^i/Z^1}}_{+1}
\eeqa
\qed
