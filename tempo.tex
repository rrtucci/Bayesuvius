\subsection{Integration, Stokes' Theorem}
Let

\beq
\ul{d^2 x}= \ul{dx}\wedge\ul{dy}
\eeq

\beq
\vecr = x\hat{x} + y\hat{y}, \quad
d\vecr =\Delta x \hat{x}
+ \Delta y \hat{y}
\eeq
and

\beq
\ul{\omega} = f(\vecr)
\ul{d^2 x}(\vecr)
\eeq
Then we define

\beq
\int \ul{\omega}\eqdef
\int
f(\vecr)\ul{d^2 x}(d\vecr)
=\int f(\vecr)\Delta x\; \Delta y
\eeq
Note that $\vecr$ 
is used as the argument
of $f$ and $d\vecr$ as
the argument of $\ul{d^2 x}$.
Similarly, one can define

\beq
\ul{d^nx}= 
\ul{dx^1}
\wedge
\ul{dx^2}
\wedge
\cdots
\wedge
\ul{dx^p}
\eeq



Change of variables

Suppose 

\beq
x=(x^1, x^2, \ldots, x^n),\quad
u=(u^1, u^2, \ldots, u^n)
\eeq

\beq
\pback{x}{x(u)} f(x) = f(x(u)),
\quad
\pback{x}{x(u)}
\ul{dx^n}(x)=
J
\ul{du^n}(u)
\eeq
where $J$ is the Jacobian

\beq
J=
\pder{x(u)}
{u}
\eeq
If 

\beq
\ul{\omega} = f
\;
\ul{d^n x}
\eeq

\beq
\int \ul{\omega}=
\int \ul{d^n x} f(x)=
\int J\;\ul{d^nu}(u)\; f(x(u))=
\int \pback{x}{ x(u)}\ul{\omega}
\eeq

Stokes' theorem
\beq
\int_U d\pform{\omega}{p}
=
\int_{\partial U}\pform{\omega}{p}
\label{eq-stokes-dif-forms}
\eeq

\beq
\underbrace{\iint\cdots\int}_{\text{$p+1$ times}} d \pform{\omega}{p}
=
\underbrace{\iint\cdots\int}_{\text{$p$ times}}\pform{\omega}{p}
\eeq


\begin{claim}
Stokes' Theorem
Eq.(\ref{eq-stokes-dif-forms}) implies

\beq
\iint_Sd\vec{S}\cdot
\nabla\times \vec{F}
=
\oint_C d\vecr \cdot\vec{F}
\quad
\text{(Curl Theorem)}
\eeq

\beq
\iiint_V\; dV
\nabla\cdot \vec{F}=
\iint_{\partial V}
d\vec{S}\cdot
\vec{F}
\quad\text{(Divergence Theorem)}
\eeq


\end{claim}
\proof

Let

\beq
\vec{F}= [F_1, F_2, F_3], \quad
\ul{\omega}=
F_1 \ul{dx}
+ F_2 \ul{dy}
+
F_3 \ul{dz}
\eeq


\begin{itemize}

\item Curl Theorem

\beq
\ul{d\omega}=
[\nabla \times \vec{F}]_1 \ul{dy}\wedge
\ul{dz}
+ \cpermu{123}
\eeq
Thus, applying Stokes'
theorem
to $d\ul{\omega}$ leads to
the Curl
Theorem.

\item Divergence Theorem

\beq
\star\ul{\omega}=
F_1 \ul{dy}\wedge\ul{dz}
+
F_2 \ul{dz}\wedge\ul{dx}
+
F_3 \ul{dx}\wedge\ul{dy}
\eeq

\beq
d\star\ul{\omega}=
\nabla\cdot \vec{F}
\ul{dx}\wedge
\ul{dy}\wedge
\ul{dz}
\eeq
Thus, applying Stokes'
theorem
to $d\star\ul{\omega}$ leads to
the Divergence
Theorem.

\end{itemize}

\qed
