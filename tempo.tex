\section{Lie Groups}

We will assume
the reader is familiar with
the mathematical
concepts of a vector space, Group,
Algebra and Lie algebra.
For some background on those,
see the beginning chapters
of my book, complementary to this one, 
Bayesuvius Quantico ********%\cite{}
What we aim to
do in this section
is to describe
how Lie Groups
can be interpret 
using the
tools and language
of differential geometry 
and p-forms.

\subsection{All tangent spaces are
isomorphic}


\begin{figure}[h!]
\centering
\includegraphics[width=3.5in]
{differ-geo/lie-group-manifold.png}
\caption{Pictorial representation of
the Lie Group manifold $G$.
All tangent spaces are identical 
copies of $T_e(G)=\ger{g}$
}
\label{fig-lie-group-man}
\end{figure}



$G=$ Lie group. This will be 
the manifold $M$
in the
differential geometry
interpretation.

$\ger{g}=$ Lie algebra for group $G$.

$e=$ identity of $G$

$T_e (G)=$ tangent space at point $e$. We will
equate $T_e(G)$ with $\ger{g}$

$T_g (G)=$ tangent space at point $g$

 

$\{E_\alp\}_{\alp=1}^n=$  generators of Lie algebra


\beq
g= \exp(t^\alp E_\alp)
\eeq
where $g\in G$, $t_\alp\in\RR$.



The various points $g$ of the manifold
are different, but the tangent spaces
$T^L_g$ for  each $g\in G$ 
are identical copies of each other
(i.e., isomorphic).
Remember, when we speak of a Lie algebra
in a non-differential geometry context, 
the elements of the Lie Algebra
are not labelled by a point $g$. They
are just a set of matrices (if a matrix representation
is being used, which is almost always the case).
So we first spend a  lot of ink 
showing that we need only consider the tangent space
at $e$. 
Actually, because groups are non-Abelian (i.e.,
its elements don't commute in general),
when $M$ is a group $G$, one can define naturally at least
two isomorphic
tangent spaces at each $g\in G$, the left-invariant
tangent space $T^L_g(G)$ and the right
invariant  tangent space $T_{g_R}(G)$.
We usually set $T^L_g=T^L_g(G)$
early on,
and ignore $T_{g_R}(G)$ thereafter.
After we are finished expressing a 1-1 onto
map of the tangent vectors $\vecX_e$ at $e$ 
and the (Left Invariant) tangent vectors $\vecX_g$ at $g$, we will  
talk of tangent vectors fields $\vecX$ which are 
defined for all $g\in G$ (i.e., 
they are
functions that map $g\mapsto T^L_g$)


Define {\bf left $(g)_L$ and right $(g)_R$
 translations by $g$} by
\beq
(g)_L h = gh, \quad (g)_R h = hg
\eeq

A {\bf left invariant vector field (LIVF)} is
a vector field $\vecX^L$ such that for all $g\in G$, $\vecX_{g}^L\in T^L_g (G)$
 is defined  from $\vecX_{e}\in  T_e(G)$, as follows.

For any path $\gamma$ such that

\beq
\gamma:[0,1]\rarrow G, \quad
\gamma(0)=e, \quad
\vecdot{\gamma}(0)=\vecX_{e}
\eeq
define the map $\dL{g}{e}$ by

\beq
\dL{g}{e}: T_e(G)\rarrow T^L_g(G)
,\quad
\dL{g}{e}\vecX_{e}= \AT{\frac{d\;}{d\tau}}{\tau=0} g\gamma(\tau)
\label{eq-LIVF-def}
\eeq
Note that the path $\gamma(\tau)\in G$ is
arbitrary except for what it does in the
neighborhood of $e$.
Translation is
very different from parallel
transport which we will
discuss when we define
covariant derivatives.
For parallel transport 
you need to be able to
measure the angle
between a curve and a vector, 
which means you need a \qt{connection}.
After a LIVF translation,
$\vecX_e$ and $\vecX_g^L$ need not
be parallel translations of each other.
The relative angle between them is indeterminate.

A {\bf right invariant vector field (RIVF)} is
a vector field $\vecX^R$
is defined in the same way as a
LIVF $\vecX^L$ except that
Eq.(\ref{eq-LIVF-def}) is replaced by

\beq
\dR{g}{e}: T_e(G)\rarrow T^R_g(G)
,\quad
\dR{g}{e}\vecX_{e}= \left(\AT{\frac{d\;}{d\tau}}{\tau=0} \gamma(\tau)\right)g
\label{eq-RIVF-def}
\eeq
Note that $\vecX^R_g\neq \vecX^L_g$
except when $g=e$.
This is easy to see in
the matrix representation wherein

\beq
X^L_g=g \vecX_e, 
\quad X^R_g = \vecX_e g
\eeq
so 

\beq
X^L_g = g X^R_g  g^{-1}
\eeq
From here on, we 
will use the LIVF almost
exclusively and drop
most of the $R$ and $L$ labels
from $\vecX^R$
and $\vecX^L$. If 
we ever need to deal with
both LIVF and RIVF at
the same time, 
we will revert 
to using the $R$ and $L$
labels.

For a LIVF $\vecX$, 
for all $g\in G$,
$\vecX_g$ is defined from $\vecX_e$ by

\beq
\vecX_g = \dL{g}{e}\vecX_e
\label{eq-dl-def}
\eeq
This map can be
extended to
\beq
\dL{g}{a}
\vecX_{a} \eqdef
\vecX_{ga}
\eeq
for any $a\in G$.
This is in fact why these
tangent vectors are called
left invariant: left translating
a vector of the vector field
produces another vector of the
vector field. Note that from Eq.(\ref{eq-LIVF-def}),
if matrix representation are allowed,
$\dL{g}{a}$ acts like  $(g)_L$

\beq
\dL{g}{a}
\vecX_{a} \eqdef
\vecX_{ga}
=
g\vecX_a
\eeq

For a Lie group,\footnote{$\cong$ means isomorphic, i.e., there exists a 1-1 onto map between the two sets,
and operations are preserved too.} let
\begin{mybluearea}
\begin{align}
T_e(G)&=  \{ \vecX_e=\vecdot{\gamma}(0)\mid
\text{$\gamma:[0,1]\rarrow G$ is a path with $\gamma(0)=e$} \}
\\
&= \{ \vecX_e(\xi)\mid \xi\}\quad \text{if the different
starting directions are parametrized by $\xi$}
\nonumber
\\
&\text{where the set of $\xi$ is isomorphic to $\ger{g}$, so we can say $\xi\in \ger{g}$}
\end{align}

\beq
T^L_g(G) = \{\vecX_g(\xi)\mid \xi\}
\eeq

\beq
T^L(G)=\sqcup_{g\in G}T^L_{g}(G)
=
\{\vecX_g(\xi)\mid g, \xi\}=\text{LIVF  tangent bundle}
\eeq

\beq
\ger{X}^L=\{\vecX(\xi)\mid \xi\}=\text{set of LIVF} 
\quad\quad\text{ where }
\vecX(\xi): G\rarrow T^L(G), \quad
g\mapsto \vecX_g(\xi)
\eeq
\end{mybluearea}
Then
\beq
\ger{g}=T_e(G)\cong T^L_{g}(G)\cong \ger{X}^L(G)
\eeq

Define 

\beq 
T^{Lp}_g(G)= \underbrace{T^L_g(G)\times\cdots\times T^L_g(G)}_{\text{$p$ copies}},
\quad
\wedge^{Lp}_g(G)= \underbrace{\wedge^L_g(G)\wedge\cdots\wedge \wedge^L_g(G)}_{\text{$p$ copies}}
\eeq
where
$\wedge^L_g(G)$ is the set of 
of left invariant 1-forms to
be defined next.

For 
$\ul{\omega_a}:\wedge^{Lp}_a(G)\rarrow \RR$
and $\vecX_a\in T^{Lp}_a(G)$,
define the following pullback


\beq
\pback{a}{ga}{\ul{\omega_a(\vecX_a)}}=
\ul{\omega_{ga}}(\vecX_{ga})
\eeq

{\bf Left invariant p-forms (LIPF)}
$\ul{\omega_a}$ satisfy

\beq
\pback{a}{ga}{\ul{\omega_a(\vecX_a)}}=
\ul{\omega_{ga}}(\vecX_{ga})=\ul{\omega_a}(\vecX_a)
\eeq
In particular, if $a=e$,

\beq
\ul{\omega_g}(\vecX_g)=\ul{\omega_e}(\vecX_e)
\label{eq-LI-p-form}
\eeq
Thus,  an LIPF
is fully specified  by its value 
at each $\vecX_e\in \ger{g}$.
For LIPF, we are fully
justified in dropping the $g$
subscripts and writing $\omega(\vecX)$ for both sides
of Eq.(\ref{eq-LI-p-form}).
Note that if you compare Eq(\ref{eq-dl-def}) with
Eq.(\ref{eq-LI-p-form}),
the statement of left invariance for
a vector
field has a $\dL{g}{e}$
operator on the left side (or, in the matrix representation, a factor of $g$ on the left side), 
whereas the statement of left invariance for a p-form
doesn't have that operator on the left side. 
If we were using
right invariant vector 
fields, the factor of 
$g$ would be on the
right side.


For $\vecX_e, \vecY_e\in T_e(G) =\ger{g}$, let
$[\vecX_e, \vecY_e]$ stand for the Lie bracket, which is
assumed to be given
for the assumed Lie Algebra. Then 
for any LIVF
$\vecX, \vecY$,
define
the {\bf Left invariant Lie braket (LILB)} by

\beq
[\vecX,  \vecY]_g
\eqdef
\underbrace{
\dL{g}{e}
\underbrace{[\vecX_e, \vecY_e]}
_{\in T_e(G)}}_{\in T^L_g}
\eeq
Note that we avoid using $[\vecX_g, \vecY_g]$
due to a technicality. $[\vecX_g, \vecY_g]$
contains only info about 
$\vecX$ and $\vecY$ at $g$,
not in the neighborhood 
of $g$. The latter info 
is necessary to compute derivatives.

Suppose we define an inner 
product on the Lie algebra:

\beq
\av{\cdot, \cdot}:
\ger{g}\times\ger{g}\rarrow \RR
\eeq
Now
define
a {\bf left invariant
metric (LIM)} $\metric^L_g$ by

\beq
\metric_g^L(\vecX_g^L, \vecY_g^L)=\av{\vecX_e, \vecY_e}
\eeq
One can a.l.so
define a {\bf
right invariant metric (RIM)} $\metric^R_g$ by

\beq
\metric_g^R(\vecX_g^R, \vecY_g^R)=\av{\vecX_e, \vecY_e}
\eeq 
and a {\bf bi-invariant metric (BIM)}
$\metric^B_g$ by


\beq
\metric_g^B(\vecX_g^L, \vecY_g^L)=
\av{\vecX_e, \vecY_e}
=
\metric_g^B(\vecX_g^R, \vecY_g^R)
\eeq
This distinction becomes very important for the covariant derivative.

How 3 derivatives
$d$, $\nnabla_\vecX$ and $\call_\vecX$
specialize 
for Lie Groups and LIVF

\begin{enumerate}
\item exterior derivative $d$:

If $\pform{\alp}{}\in \wedge^L(G)$
and $\vecX, \vecY\in\ger{X}^L$,

\beq
\ul{d\alp_g}
(\vecX_g, \vecX_g)
=
\vecX_g[\ul{\alp_g}(\vecY_g)]
-
\vecY_g[\ul{\alp_g}(\vecX_g)]
-\ul{\alp_g}
([\vecX, \vecY]_g)
\eeq

\beq
\vecX_g[
\ul{\alp_g}(\vecY_g)]
]
=
\vecX_g[\underbrace{
\ul{\alp_e}(\vecY_e)}_{\text{real number}}]
=0
\eeq

\beq
\vecY_g[\ul{\alp_g}(\vecX_g)]
=
\vecY_g[\ul{\alp_e}(\vecX_e)]
=0
\eeq

\beq
\ul{\alp_g}
([\vecX, \vecY]_g) = 
\ul{\alp_e}
([\vecX_e, \vecY_e])
\eeq

\beq
\boxed{
\ul{d\alp_g}
(\vecX_g, \vecX_g)
=
-\ul{\alp_e}
([\vecX_e, \vecY_e])}
\eeq

\item Lie derivative $\call_\vecX$:

\beq
\boxed{\call_{\vecX_g}\vecY_g=
[\vecX, \vecY]_g =
\dL{g}{e} [\vecX, \vecY]_e}
\eeq

\item Covariant derivative $\nnabla_\vecX$:


\beq
\nnabla_{\vecX_g}\vecY_g=
\ul{(\omega\indices{^\rho_\mu})_g(\vecX_g)}\vecY_g=
\ul{(\omega\indices{^\rho_\mu})_e(\vecX_e)}\vecY_g
\eeq

\beq
\nnabla_{\vecX_g}\vecY_g=
\dL{g}{e}
\nnabla_{\vecX_e}\vecY_e
\eeq



\end{enumerate}

\subsection{Killing Vector Fields}

Wilhelm Killing

\begin{claim}


\beq
[\vecX^L_g, \vecY^L_g] =
g[\vecX_e, \vecY_e]
\eeq

\beq
[\vecX^R_g, \vecY^R_g] =
-[\vecX_e, \vecY_e]g
\eeq

\beq
[\vecX^R_g, \vecY^L_g] =
[\vecX^L_g, \vecY^R_g] =0
\eeq

\end{claim}
\proof



Assume a matrix representation

\beq
\vecX_g^L = g\vecX_e,
\quad
\vecX_g^R = \vecX_e g
\eeq

\beq
\oper{X^R_{g}}= \AT{\frac{d}{dt}}{t=0}(\oper{g(t;\vecX_e)})_L,
\quad
\oper{X^L_{g}}= \AT{\frac{d}{dt}}{t=0}(
\oper{g(t;\vecX_e)}
)_R
\eeq
where 

\beq
\oper{g(t;\vecX_e)}=
e^{t\oper{\vecX_e}}
\eeq


\beqa
[\vecX^L_g, \vecY^L_g]
&=&
\oper{\vecX^L_g}(\vecY^L)-
(X\swap Y)
\\
&=&
 \AT{\frac{d}{dt}}{t=0}(g(t;\vecX_e))_R(g\vecY_e)
-
(X\swap Y)
\\
&=&
g\oper{\vecX_e}(\vecY_e)
-
(X\swap Y)
\\
&=&
g[\vecX_e, \vecY_e]
\eeqa


\beqa
[\vecX^R_g, \vecY^R_g]
&=&
\oper{\vecX^R_g}(\vecY^R)-
(X\swap Y)
\\
&=&
 \AT{\frac{d}{dt}}{t=0}(g(t;\vecX_e))_L(\vecY_eg)
-
(X\swap Y)
\\
&=&
\oper{\vecY_e}(\vecX_e)g
-
(X\swap Y)
\\
&=&
-[\vecX_e, \vecY_e]g
\eeqa


\beqa
[\vecX^R_g, \vecY^L_g]
&=&
\oper{\vecX^R_g}(\vecY^L)-
(XR\swap YL)
\\
&=&
 \AT{\frac{d}{dt}}{t=0}(g(t;\vecX_e))_L(g\vecY_e)
-
 \AT{\frac{d}{dt}}{t=0}(g(t;\vecY_e))_R(\vecX_e g)
\\
&=&
\oper{\vecX_e}(g\vecY_e)
-
\oper{\vecX_e}(g\vecY_e)
\\
&=&
0
\eeqa
\qed

Consider Eq.(\ref{eq-lie-deriv-metric})
which we repeat:

\beq
\call_\vecX \metric(\vecU, \vecV)
=
\vecX(\metric(\vecU, \vecV))
-
\metric([\vecX, \vecU], \vecV)
-
\metric(\vecU, [\vecX, \vecV])
\label{eq-lie-deriv-metric-copy1}
\eeq

$\vecX$ is a Killing vector field if $\call_\vecX \metric =0$
 

\begin{claim}
Left invariant
vector fields are 
Killing fields of
a right invariant metric
and vice-versa

\beq
\call_{\vecX^R_g}\metric_g(\vecU^L_g, \vecV^L_g)=
\call_{\vecX^L_g}\metric(\vecU^R_g, \vecV^R_g)=0
\eeq
Also, in general,
left invariant
vector fields are not
Killing fields of
a left invariant metric
and the same
statement with left replaced by right.
\beq
\call_{\vecX^R_g}\metric_g(\vecU^R_g, \vecV^R_g)\neq 0, \quad
\call_{\vecX^L_g}\metric(\vecU^L_g, \vecV^L_g)\neq 0
\eeq
\end{claim}
\proof

Since $\metric$ is assumed
left invariant,

\beq
\metric_g(\vecU^L_g, \vecV^L_g)=
\metric_e(\vecU_e, \vecV_e)
\eeq
This is a constant as far as $\vecX$ is concerned so

\beq
\vecX(\metric_g(\vecU^L_g, \vecV^L_g))=0
\eeq
The 
first term
on the right side
of 
Eq.(\ref{eq-lie-deriv-metric-copy1})
is zero  so


\beq
\call_{\vecX^R_g} \metric_g(\vecU^L_g, \vecV^L_g)
=
\metric_g(
\underbrace{[\vecX^R_g, \vecU^L_g]}_0
, \vecV^L_g)
+
\metric_g(\vecU^L_g, 
\underbrace{[\vecX^R_g, \vecV^L_g]}_0
)=0
\eeq

\beq
\call_{\vecX^L_g} \metric_g(\vecU^L_g, \vecV^L_g)
=
\metric_g(
\underbrace{[\vecX^L_g, \vecU^L_g]}_
{g[\vecX_e, \vecU_e]}
, \vecV^L_g)
+
\metric_g(\vecU^L_g, 
\underbrace{[\vecX^L_g, \vecV^L_g]}_
{g[\vecX_e, \vecV_e]}
)\neq 0
\eeq


\beq
\call_{\vecX^R_g} \metric_g(\vecU^R_g, \vecV^R_g)
=
\metric_g(
\underbrace{[\vecX^R_g, \vecU^R_g]}_
{-g[\vecX_e, \vecU_e]}
, \vecV^R_g)
+
\metric_g(\vecU^R_g, 
\underbrace{[\vecX^R_g, \vecV^R_g]}_
{-g[\vecX_e, \vecV_e]}
)\neq 0
\eeq



\qed


\subsection{Exp, Adj and adj maps}

In this section, we assume $G$
has a matrix representation.

For every $\vecX\in \ger{g}$ define
the {\bf exponential map} $\exp(\cdot)$ by

\beq
\exp: \ger{g}\rarrow G, \quad
\exp(\vecX) = 1 + \vecX + \frac{1}{2}(\vecX)^2 + \ldots
\eeq

Note that
\beq
g(\tau) = \exp(t \vecX)
\eeq
is a path in $G$. Since

\beq
e^{t_1 \vecX}e^{t_2\vecX}=
e^{(t_1+t_2)\vecX}
\eeq
this path is a subgroup of $G$.

For every $\vecX\in \ger{g}$, define
the {\bf large A adjoint  map} $\Ad(\;)$ by

\beq
\Ad_g:\ger{g}\rarrow \ger{g}, \quad
\Ad_g \vecX = g\vecX g^{-1}
\eeq

For every $\vecX\in \ger{g}$, define
the {\bf small A adjoint map}  $\ad(\;)$ by

\beq
\ad_\vecX:\ger{g}\rarrow \ger{g}, \quad
\ad_\vecX \vecY = [\vecY, \vecX]
\eeq


\begin{claim}

For any $n\times n$ matrices $A$ and $B$,

\beq
e^{A}B e^{-A}=
e^{[A,\cdot ]}B
\eeq
In particular, if $A=\vecX$
\beq
\Ad_{\exp\vecX}=
\exp(\ad_\vecX)
\eeq

\end{claim}
\proof

Define $M(t)$ by
\beq
M(t)=e^{tA}Be^{-tA}
\eeq
Note that

\beq
M(0)=B
\eeq
and 
\beq
M'(t)= AM(t)- M(t)A= [A, M(t)]
\eeq
Think of $[A, \cdot]$ as an operator.
Clearly, this differential equation
and initial condition is satisfied 
by $M(t)=e^{t[A,\cdot]}B$. Now set $t=1$ to get

\beq
e^{A}Be^{-A}= e^{[A,\cdot]}B
\eeq

\qed


\subsection{Maurer-Cartan equation}
Let $G$ be a Lie group and $\ger{g}= T_e(G)$.
The {\bf Maurer-Cartan 1-form}
$\pform{\vTheta}{}$ is defined as 
follows.


\beq
\maurer{g}: T^L_g(G)\rarrow T_e(G),\quad \vecX_g \mapsto \dL{g^{-1}}{g}\vecX_g = \vecX_e
\eeq

Note that $\maurer{g}: T^L_g(G)\rarrow T_e(G)$
whereas $\dL{g}{e}: T_e(G)\rarrow T^L_g(G)$
so they go in opposite directions. 
In fact, $\maurer{g}$ and 
$\dL{g^{-1}}{g}$ are inverse of each other:

\beq
\dL{g}{e}\vecX_e = \vecX_g,
\quad 
\maurer{g} \vecX_g =\vecX_e
\eeq

Note that $\maurer{}$  is a 1-form whose image
is a tangent vector at $e$ (rather than a real number, like 
the usual 1-forms). That is the reason why
we have chosen to do both, underline it and give it a
vector sign. We have actually
encountered vector valued 1-forms before, in $\ul{d\vecr}$.
In fact, $\maurer{g}$ plays the same
role for Lie Groups as $\ul{\vec{dr}}$
does in the generic
Cartan theory. Both are 
expanded as $\theta^a \vece_a$.
Both have a \qt{beginning} ($e$) and an \qt{end} ($g$)

\begin{claim}
For any matrix $g(\tau)\in G$
\beq
d(g^{-1})=
-g^{-1}dg g^{-1}
\eeq

\beq
gd(g^{-1})=
- g^{-1}dg
\eeq
\end{claim}
\proof

\beq
0=d(g^{-1}g)=
d(g^{-1})g + g^{-1}dg
\eeq
\qed

\begin{claim}
For a matrix Lie group
(or a Lie group with a matrix representation) and $g\in G$

\beq
\maurer{g}\vecX_g= g^{-1}\ul{dg}
\eeq

\end{claim}
\proof

Suppose

\beq
\gamma:[0,1]\rarrow G, \quad
\gamma(0)=g,
\quad \vecdot{\gamma}(0)=\vecX_g
\eeq
Then

\beq
\maurer{g} \vecX_g=
\dL{g^{-1}}{g}\vecX_g=
g^{-1} \vecX_g\cdot \nabla
 \gamma(0)
= g^{-1}\ul{dg}(\vecX_g)
\eeq
\qed

What is $\ul{dg}$?

If
\beq
g = [g_{\mu, \nu}]^{n\times n}
\eeq
then

\beq
\ul{dg}=
[\ul{dg_{\mu, \nu}}]^{n\times n}
\eeq
so $\ul{dg}$ is an $n\times n$ matrix of 1-forms. For
$n=2$, if $\{\ul{dx}, \ul{dy}\}$
is a coordinate dual t-basis
of $G$, 
then

\beq
\ul{dg}=A\ul{dx}+ B\ul{dy}
\eeq
where $A$ and $B$ are $2\times 2$ matrices.

It's always true that
$\ul{\alp}\wedge \ul{\alp}=0$
for ordinary 1-forms. For example,
for ordinary 1-forms in $n=2$, $\ul{\alp}=a \ul{dx}+
b\ul{dy}$
with scalar coefficients $a, b$,
so $\ul{\alp}\wedge \ul{\alp}=0$.
 However,
$\ul{dg}$ is not an ordinary 1-form: it's a
1-form with matrix coefficients which don't commute in general.
For example, for $n=2$,

\beq
\ul{dg}= A \ul{dx}+ B\ul{dy}
\eeq
so 
\beq
\ul{dg}\wedge \ul{dg}=
[A, B]\ul{dx}\wedge \ul{dy}
\eeq





Choose a basis $\{\vecE_\alp\}_{\alp=1}^n$ for $\ger{g}$. Assume

\beq 
[\vecE_\beta, \vecE_\gamma]= C\indices{^\alp_{\beta\gamma}}
\vecE_\alp
\eeq

Expand 

\beq
\pform{\vTheta}{}=
\pform{\Theta^\alp}{}\vecE_\alp
\eeq
Assume

\beq
\ul{\Theta^\alp}(\vecE_\beta)=
\delta^\alp_\beta
\eeq


\begin{claim}

\beq
\ul{d\Theta^\alp}(\vecE_{\mu}, \vecE_\nu)=
-C\indices{^\alp_{\mu\nu}}
\eeq
\end{claim}
\proof


Let

\beq

\subsection{Cartan Structure Equations for Lie Group}

The Maurer-Cartan equation is intrinsic to the group; it's independent of the connection.
The Cartan structure equations,
in the other hand, assume a connection.
To define 
define $\pform{\omega}{}$, you must choose a connection—most commonly the Levi-Civita connection of a chosen left-invariant Riemannian metric.

Let $G$ be an
$n$ dim Lie group
with $\ger{g}=
T_e(G)$.
Choose a basis $\vecE_\mu$ (these are  left invariant vector fields that map $g\in G$ to
$\vecE_\mu(g)$) of $\ger{g}$
such that

\beq
[\vecE_\mu, \vecE_\nu]=
C\indices{^\rho_{\mu\nu}}
\vecE_\rho
\eeq
Expand 

\beq
\pform{\vTheta}{}=
\pform{\Theta^\alp}{}\vecE_\alp
\eeq
Assume

\beq
\ul{\Theta^\alp}(\vecE_\beta)=
\delta^\alp_\beta
\eeq

Recall we 
proved

\beq
\ul{d\Theta^\alp}=-\;\frac{1}{2}
C\indices{^\alp_{\mu\nu}}
\ul{\Theta^\mu}
\wedge
\ul{\Theta^\nu}
\eeq
Introduce a left invariant \Rien metric 
 $G$
such that the
$\vecE_\mu$
are orthogonal in that metric.

Introduce LCC
given by



\beq
\nnabla
\vecE_\nu =
\ul{\omega\indices{^\rho_\mu}}\vecE_\rho
,\quad
\nnabla_\vecX
\vecE_\nu =
\ul{\omega\indices{^\rho_\mu}}(\vecX)\vecE_\rho
\eeq

\beq
\vecX= X^\nu \vecE_\nu,
\eeq

\beqa
\nnabla_{\vecE_\mu}
\vecE_\nu &=&
X^\nu
\ul{\omega\indices{^\rho_\mu}}(\vecE_\nu)
\vecE_\rho
\\
&=&
X^\nu
\Gamma\indices{^\rho_{\mu\nu}}\vecE_\rho
\eeqa

\beq
\ul{\omega\indices{^\rho_\mu}}(\vecX)=
X^\nu 
\Gamma\indices{^\rho_{\mu\nu}}
=
\Gamma\indices {^\rho_{\mu\nu}}\ul{\Theta^\nu}(\vecX)
\eeq


\begin{enumerate}
\item Cartan's first structure equation (SE 1)
\beq
\pform{T^\rho}{2}=
\ul{d\Theta^\rho}
+
\ul{
\omega\indices{^\rho_\mu}}
\wedge \ul{\Theta^\mu}
\eeq
Combining SE 1 with the Maurer-Cartan equation, we get

\beq
\pform{T^\rho}{2}
=
-\;\frac{1}{2}
C\indices{^\alp_{\mu\nu}}
\ul{\Theta^\mu}
\wedge
\ul{\Theta^\nu}
+
\ul{
\omega\indices{^\rho_\mu}}
\wedge \ul{\Theta^\mu}
\eeq

In LCC, $\pform{T^\rho}{2}=0$

\beq
\ul{
\omega\indices{^\rho_\mu}}
\wedge \ul{\Theta^\mu}
=
\frac{1}{2}
C\indices{^\alp_{\mu\nu}}
\ul{\Theta^\mu}
\wedge
\ul{\Theta^\nu}
\eeq

\item Cartan's second structure equation (SE 2)

\beq
\pform{\Omega\indices{^\alp_\beta}}{2}
=
\ul{d\omega
\indices{^\alp_\beta}}
+
\ul{\omega\indices{^\alp_\gamma}}
\wedge
\ul{\omega\indices{^\gamma_\beta}}
\eeq
\end{enumerate}

\begin{claim}
If we assume LCC, then
\beq
2\Gamma_{\gamma\beta\alp}=
C_{\gamma\alp\beta}
-C_{\alp\beta\gamma}
+C_{\beta\gamma\alp}

\end{claim}

If we assume LCC, then
Koszul's formula (see

\vecE_\beta,
\vecE_\gamma}

\av{
[\vecE_\alp,
\vecE_\beta],
\vecE_\gamma}
-(\beta\gamma\alp)
+(\gamma\alp\beta)
\eeq

\beq
2
\Gamma_{\gamma\alp\beta}
=
C_{\gamma\alp\beta}-C_{\alp\beta\gamma}
+
C_{\beta\gamma\alp}
\eeq

\qed
