Hopf  fibration

\beq
S^3 = \{(\zp, \zm)\in \CC^2 :
|\zp|^2 + |\zm|^2 =1\}\eeq

\beq
S^2 =\{(x,y,z)\in\RR^3 : x^2 + y^2 + z^2 =1
\}
\eeq



\beq
{h}: S^3:\rarrow S^2,
\quad
{h}(\zp, \zm)
=\begin{pmatrix}
x\\y\\z
\end{pmatrix}
=
\begin{pmatrix}
2Re(\zp\zm^*)
\\
2Im(\zp \zm^*)
\\
|\zp|^2 -|\zm|^2
\end{pmatrix}
\eeq


Note immediately that

\beqa
x^2 + y^2 +z^2 
&=&4|\zp\zm^*|^2
+ (|\zp|^2 -|\zm|^2)^2
\\
&=&
(|\zm|^2 + |\zp|^2)^2
\\
&=&
1
\eeqa
so $(x,y,z)\in S^2$ as advertised.

\beq
|\zp + \zm|^2 =
\underbrace{|\zp|^2 + |\zm|^2}_1 + \underbrace{\zp\zm^* +\zp^* \zm}_
{=2Re(\zp\zm^*)=x}
\eeq


\beq
|\zp + i\zm|^2 =
\underbrace{|\zp|^2 + |\zm|^2 }_1
+\underbrace{(-i\zp \zm^* +i\zm\zp^*)}_{=
  2Im(\zp\zm^*)=y}
\eeq
(sign check: $\zm=1/\sqrt{2}$, $\zp=i/\sqrt{2}$ gives $|\zp+i\zm|^2=2$ and $1+y=2$ )


\beq
z= |\zp|^2-|\zm|^2,
\quad 1=|\zp|^2 + |\zm|^2
\implies
|z_\pm|^2 = \frac{1\pm z}{2}
\eeq
