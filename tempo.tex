\subsection{Curvilinear Coordinates}

Rectangular coordinates $(x^i)_{i=1}^n$

Orthogonal Curvilinear coordinates (OCC)
$(\phi^\mu)_{\mu=1}^n$

\begin{itemize}

\item metric

\beq
ds^2 = \sum_{\mu=1}^n (\ell^\mu d\phi^\mu)^2
\eeq

\beq
\metric_{\mu\nu}=
(\ell^\mu)^2 \delta_{\mu\nu}
\quad \text{(orthogonal-non-normal t-basis)}
\eeq

\beq
(\metric^{-1})_{\mu\nu}=\metric^{\mu\nu}=
\frac{1}{(\ell^\mu)^2}
\delta_{\mu\nu}
\eeq

\beq
\metric = [\metric_{\mu\nu}],
\quad
\det \metric = \prod_{\mu=1}^n
(\ell^\mu)^2
\eeq
\beq
\ul{\theta^\mu}=\ell^\mu \ul{d\phi^\mu}
\eeq

\beq
(ds^2) =\sum_\mu (\theta^\mu)^2
\eeq

\beq
\hat{\metric}_{\mu\nu}=
\hat{\metric}^{\mu\nu}
=
\delta_{\mu\nu}
\quad(\text{orthonormal t-basis})
\eeq

\item engineering dimensions
\beq
\vecr= x^i\hat{x_i}
\eeq

\beq
\engdim{\vecr}=\engdim{x^i} =L,
\quad\quad
\engdim{\hat{x^i}} =
\engdim{\vecpder{x^i}}
=\engdim{\pder{\vecr}{x^i}}
= L^0
\eeq


\beq
\engdim{\phi^\mu} =L^0,
\quad\quad
\engdim{\vecpder{\phi^\mu}}
=\engdim{\pder{\vecr}{\phi^\mu}}
= L
\eeq

\beq
\engdim{\ell^\mu} =L,
\quad\quad
\engdim{\hat{\phi}_\mu} =
\engdim{\frac{1}{\ell^\mu}\vecpder{\phi^\mu}}
=\engdim{\frac{1}{\ell^\mu}\pder{\vecr}{\phi^\mu}}
= L^0
\eeq
Unit vectors always dimensionless.
For a scalar dimensionless function $f$ 
\beq
\engdim{(\nabla f)_\mu}=
\engdim{\pder{f}{x^\mu}}=
\engdim{\frac{1}{\ell^\mu}\pder{f}{\phi^\mu}}= L^{-1},
\quad 
\engdim{\vecpder{x^\mu}}=
\engdim{\frac{1}{\ell^\mu}\vecpder{\phi^\mu}} = L^0
\eeq
so the gradient of $f$ and its 
$x^\mu$ derivative have dimensions of 
$L$, whereas the tangent 
vector $\vecpder{x^\mu}$ is dimensionless.

\beq
\engdim{\metric_{\mu\nu}}= L^2,
\quad
\engdim{\metric^{\mu\nu}}= L^{-2}
\eeq


\item Operator $\nabla$ has multiple meanings

In rectangular coordinates $x^\mu$

\beq
\nabla^{grad}=\nabla^{div}=\nabla^{curl}
\eeq
In OCC coordinates $\phi^\mu$,
 
\beq
\nabla^{grad}, \nabla^{div}, \nabla^{curl}\quad \text{are all different}
\eeq
although they
all still have 
engineering 
dimension $L^{-1}$

\item Previous results from Section
\ref{sec-expressing-grad-div-curl}

\beq
\bcen
\xymatrix{
f\ar[d]_{\sharp\;d}
\\
\nabla  f
}\ecen
\quad\quad
\boxed
{\nabla f =
(\sharp \;d) (f)}
\eeq

\beq
\bcen
\xymatrix{
(\vecV)^\flat\ar[d]_{\sharp \star d}
\\
\nabla \times \vecV
}\ecen
\quad\quad
\boxed
{\nabla \times \vecV=
(\sharp \star  d \; \flat)(\vecV)}
\eeq

\beq
\bcen
\xymatrix{
(\vecV)^\flat\ar[d]_{\star d \star}
\\
\nabla \times \vecV
}\ecen
\quad\quad
\boxed{
\nabla \cdot \vec{V}=
(\star
d
\star
\flat)
\vec{V}
}
\eeq

\item Sharp and flat operator are linear 
and retain the index
position 
in going from a vector to covector, but 
you should  lower or raise that index with the metric

\beq
(\ul{dx^\mu})^\sharp=
\partial^\mu=
g^{\mu\nu}\partial_\nu
\eeq

\beq
(\partial_\mu)^\flat=
\ul{dx_\mu}=
g_{\mu\nu}\ul{dx^\nu}
\eeq
This distinction
is unimportant
for Cartesian coordinates but 
becomes crucial
for curvilinear
coordinates.



\end{itemize}

\begin{claim}
For OCC, if $f$
is a scalar unction, 
\beq
\nabla^{grad} f = 
\frac{\hat{\phi_\mu}}{\ell^\mu}
\pder{f}{\phi^\mu}
\eeq

\end{claim}
\proof

\beq
\pform{df}{}=
\pder{f}{\phi^\mu}
\pform{d\phi^\mu}{}
=
\frac{1}{\ell^\mu}
\pder{f}{\phi^\mu}
\ul{\theta^\mu}
\eeq

In orthonormal basis
\beq
\nabla^{grad}f=
(\ul{df})^\sharp
=
\frac{1}{\ell^\mu}
\pder{f}{\phi^\mu}
(\ul{\theta}^\mu)^\sharp
=
\frac{1}{\ell^\mu}
\pder{f}{\phi^\mu}
\hat{\phi}^\mu
=
\frac{1}{\ell^\mu}
\pder{f}{\phi^\mu}
\hat{\phi}_\mu
\eeq

In orthogonal-non-normal t-basis

\beq
(\ul{d\phi^\mu})^\sharp
=
\metric^{\mu\nu}
\vecpartial_\nu
=
\frac{1}{(\ell^\mu)^2}
\vecpder{\phi^\mu}
\eeq



\beq
\nabla^{grad}f=
\frac{1}{(\ell^\mu)^2}
\pder{f}{\phi^\mu}
\vecpder{\phi^\mu}
=
\frac{1}{\ell^\mu}
\pder{f}{\phi^\mu}
\hat{\phi}_\mu
\eeq

\qed

\begin{claim}
If $A$ is diagonalizable
with non-zero eigenvalues, then
\beq
\ln \det A = \tr \ln A
\eeq
\end{claim}
\proof

Suppose the eigenvalues
of $A$ are $\lam_i$
for $i=1, 2 \ldots,n$.
Assume $A$ is diagonal.

\beq
\ln \det A = \ln \prod_i \lam_i= \sum_i\ln \lam_i
\eeq

\beq
\tr \ln A=
\tr \;diag(\ln\lam_1,\ln \lam_2, \ldots, \ln \lam_n)=
\sum_i \ln \lam_i
\eeq

\qed


\begin{claim}
For OCC, if

\beq
\vecX = X^\mu\vecpder{\phi^\mu}=
\frac{v^\mu}{\ell^\mu}
\vecpder{\phi^\mu}
=
v^\mu \hat{\phi}_\mu
\vecv
\eeq
then

\beqa
\nabla^{div}\cdot \vec{v}&=&
\frac{1}{\sqrt{\det{\metric}}}\pder{\;}
{\phi^\mu}\left[\sqrt{\det{\metric}}
\frac{v^\mu}{\ell^\mu}\right]
\\
&=&
\frac{1}{\sqrt{\det{\metric}}}\pder{[\sqrt{\det{\metric}}X^\mu]}{\phi^\mu}
\\
&=&
\pder{X^\mu}{\phi^\mu}
+
X^\mu \pder{[\ln\sqrt{\det{\metric}}]}{\phi^\mu}
\eeqa
\end{claim}
\proof

\beq
\nabla^{div}\cdot\vec{v}=
\nnabla_\mu X^\mu
\eeq

\beq
\nnabla_\mu X^\nu=
\partial_\mu X^\nu +
\Gamma\indices{^\nu_{\mu\beta}}X^\beta
\eeq

\beq
\nnabla_\mu X^\mu=
\partial_\mu X^\mu +
\Gamma\indices{^\mu_{\mu\beta}}X^\beta
\eeq
Now use the index version of Koszul's 
identity which says:

\beq
\Gamma\indices{^\mu_{\alp\beta}}=
\frac{1}{2}
\metric^{\mu\nu}
(\partial_\alp \metric_{\nu\beta}
+(\alp\swap \beta)
-\partial_\nu\metric_{\alp\beta}
)
\eeq

\beqa
\Gamma\indices{^\mu_{\mu\beta}}
&=&
\frac{1}{2}
\metric^{\mu\nu}
(
\cancel{\partial_\mu \metric_{\nu\beta}}
+\partial_\beta \metric_{\nu\mu}
-
\cancel{\partial_\nu\metric_{\mu\beta}}
)\quad(\text{two terms cancel})
\\
&=&
\frac{1}{2}
\metric^{\mu\nu}\partial_\beta \metric_{\mu\nu}
\\
&=&
\frac{1}{2}
\partial_\beta
 \sum_{\mu} \ln g_{\mu\mu}
\\
&=&
\frac{1}{2}
\partial_\beta\tr \ln\; \metric
\\
&=&
\frac{1}{2}\partial_\beta\ln\det\;\metric
\\
&=&
\partial_\beta\ln \sqrt{\det \metric}
\eeqa



\beq
\nnabla_\mu X^\mu=
\partial_\mu X^\mu +
X^\beta
\partial_\beta\ln \sqrt{\det \metric}
\eeq

Alternative proof for $n=3$:

\beq
\star(\ul{\theta^1}
\wedge
\ul{\theta^2})=
\ul{\theta^3}
\eeq
and 2 other cyclic
permutations of 1,2,3.

\beq
\nabla \cdot \vecv=
(\star
d
\star
\flat)
\vecv
\eeq

\beqa
d(\star \flat \vecv)
&=&
d(\star v^\mu \hat{\phi}_\mu)
\\
&=&
d(v^\mu )
\eeqa



\qed

\begin{claim}
For OCC,

\beqa
\nabla^2 f &=&
\nabla^{div}\cdot \nabla^{grad}f
\\
&=&\frac{1}{\sqrt{\det{\metric}}}\pder{\;}
{\phi^\mu}\left[\sqrt{\det{\metric}}
\frac{\hat{\phi}_\mu}{(\ell^\mu)^2}
\pder{f}{\phi^\mu}\right]
\eeqa


\end{claim}
\proof

\qed

\begin{claim}
For OCC in 3 dim, if

\beq
\vecX = X^\mu\vecpder{\phi^\mu}=
\frac{v^\mu}{\ell^\mu}
\vecpder{\phi^\mu}
=
v^\mu \hat{\phi}_\mu
=\vecv
\eeq
then

\beq
\nabla^{curl}\times \vec{v}=
\frac{1}{\ell^1 \ell^2 \ell^3}
\left|
\begin{array}{ccc}
\ell^1\hat{\phi}_1
&\ell^2\hat{\phi}_2
&\ell^3\hat{\phi}_3
\\
\partial_{\phi^1}
&\partial_{\phi_2}
&\partial_{\phi_3}
\\
\ell^1 v^1
&\ell^2 v^2
&\ell^3 v^3
\end{array}
\right|
\eeq

\beq
[\nabla^{curl}\times \vecv]_1=
\frac{1}{ \ell^2\ell^3}
\hat{\phi}_1
\left[
\pder{(\ell^3 v^3)}{\phi^2}
-
\pder{(\ell^2 v^2)}{\phi^3}
\right]
\eeq
\end{claim}
\proof

\beq
\nabla \times \vecv=
(\sharp \star  d \; \flat)(\vecv)
\eeq

\beqa
(d\flat)(v^\mu \hat{\phi}_\mu)
&=&
d
(v^\mu \ul{\theta_\mu})
\\
&=&
d(v^\mu \ul{\theta^\mu})
\\
&=&
d(\ell^\mu v^\mu \ul{d\phi^\mu})
\\
&=&
\partial_\nu{(\ell^\mu v^\mu)}
\ul{d\phi^\nu}
\wedge
\ul{d\phi^\mu}
\\
&=&
\frac{1}{2}
[
\partial_\nu{(\ell^\mu v^\mu)}-
(\mu\swap \nu)]
\ul{d\phi^\nu}
\wedge

\\
&=&
\frac{1}{2\ell^\mu\ell^\nu}
[

\eeqa

\beq

\eeq
and 2 other cyclic

\beq

\frac{1}{\ell^2\ell^3}
[
\partial_2{(\ell^3 v^3)}-
(2\swap 3)]
\ul{\theta^1}
+ \cpermu{123}
\eeq

\beq
(\sharp\star d\flat)(v^\mu \hat{\phi}_\mu)
=
\frac{1}{\ell^2\ell^3}
[
\partial_2{(\ell^3 v^3)}-
(2\swap 3)]

\qed
